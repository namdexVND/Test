VNDHubUI.Tabs = {}
local CurrentTab = nil

local function CreateTab(TabName)
	-- Tab Button
	local TabButton = Instance.new("TextButton")
	TabButton.Name = TabName
	TabButton.Parent = SidebarScroll
	TabButton.BackgroundColor3 = Colors.Sidebar
	TabButton.BackgroundTransparency = 1
	TabButton.TextColor3 = Colors.TextSecondary
	TabButton.TextSize = 12
	TabButton.Font = Enum.Font.Gotham
	TabButton.Size = UDim2.new(1, 0, 0, 45)
	TabButton.BorderSizePixel = 0
	TabButton.Text = TabName
	TabButton.TextWrapped = true
	TabButton.AutomaticSize = Enum.AutomaticSize.Y

	-- Tab Content Frame
	local TabContent = Instance.new("ScrollingFrame")
	TabContent.Name = TabName .. "Content"
	TabContent.Parent = ContentScroll
	TabContent.BackgroundTransparency = 1
	TabContent.Size = UDim2.new(1, 0, 1, 0)
	TabContent.ScrollBarThickness = 0
	TabContent.CanvasSize = UDim2.new(0, 0, 0, 0)
	TabContent.Visible = false
	TabContent.BorderSizePixel = 0

	local TabContentLayout = Instance.new("UIListLayout")
	TabContentLayout.Parent = TabContent
	TabContentLayout.SortOrder = Enum.SortOrder.LayoutOrder
	TabContentLayout.Padding = UDim.new(0, 8)

	-- Tab Object
	local Tab = {}
	Tab.Name = TabName
	Tab.Button = TabButton
	Tab.Frame = TabContent
	Tab.Layout = TabContentLayout
	Tab.Elements = {}

	-- Toggle Tab
	TabButton.MouseButton1Click:Connect(function()
		for _, v in pairs(VNDHubUI.Tabs) do
			if v.Frame then v.Frame.Visible = false end
			if v.Button then
				v.Button.BackgroundTransparency = 1
				v.Button.TextColor3 = Colors.TextSecondary
			end
		end
		TabContent.Visible = true
		TabButton.BackgroundColor3 = Colors.Accent
		TabButton.BackgroundTransparency = 0.3
		TabButton.TextColor3 = Colors.Text
		CurrentTab = Tab
	end)

	-- TAB METHODS
	-- ==========================================

		ValueLabel.Size = UDim2.new(0.5, -10, 0, 20)
		ValueLabel.Position = UDim2.new(0.5, 0, 0, 8)
		ValueLabel.TextXAlignment = Enum.TextXAlignment.Right
		ValueLabel.Text = tostring(Options.Default or 0)

		local SliderBack = Instance.new("Frame")
		SliderBack.Parent = SliderContainer
		SliderBack.BackgroundColor3 = Colors.Background
		SliderBack.BackgroundTransparency = 0
		SliderBack.Size = UDim2.new(1, -20, 0, 8)
		SliderBack.Position = UDim2.new(0, 10, 0, 38)
		SliderBack.BorderSizePixel = 0

		local SliderBackCorner = Instance.new("UICorner")
		SliderBackCorner.CornerRadius = UDim.new(0, 4)
		SliderBackCorner.Parent = SliderBack

		local SliderFill = Instance.new("Frame")
		SliderFill.Parent = SliderBack
		SliderFill.BackgroundColor3 = Colors.Accent
		SliderFill.BackgroundTransparency = 0
		SliderFill.Size = UDim2.new((Options.Default - (Options.Min or 0)) / ((Options.Max or 100) - (Options.Min or 0)), 0, 1, 0)
		SliderFill.BorderSizePixel = 0

		local SliderFillCorner = Instance.new("UICorner")
		SliderFillCorner.CornerRadius = UDim.new(0, 4)
		SliderFillCorner.Parent = SliderFill

		local CurrentValue = Options.Default or 0

		SliderBack.InputBegan:Connect(function(input, gameProcessed)
			if gameProcessed then return end
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				while UserInputService:IsMouseButtonPressed(Enum.UserInputButton.Left) do
					local Mouse = LocalPlayer:GetMouse()
					local SliderSize = SliderBack.AbsoluteSize.X
					local SliderPos = SliderBack.AbsolutePosition.X
					local MousePos = Mouse.X
					local ClickPos = MousePos - SliderPos
					local Percentage = math.clamp(ClickPos / SliderSize, 0, 1)
					local Value = math.round(Percentage * ((Options.Max or 100) - (Options.Min or 0)) + (Options.Min or 0))
					CurrentValue = Value
					SliderFill.Size = UDim2.new(Percentage, 0, 1, 0)
					ValueLabel.Text = tostring(Value)
					if Options.Callback then
						Options.Callback(Value)
					end
					game:GetService("RunService").RenderStepped:Wait()
				end
			end
		end)

		return {
			GetValue = function()
				return CurrentValue
			end,
			SetValue = function(self, Value)
				CurrentValue = Value
				local Percentage = (Value - (Options.Min or 0)) / ((Options.Max or 100) - (Options.Min or 0))
				SliderFill.Size = UDim2.new(Percentage, 0, 1, 0)
				ValueLabel.Text = tostring(Value)
			end
		}
	end

		DropdownButton.Parent = DropdownContainer
		DropdownButton.BackgroundColor3 = Colors.Background
		DropdownButton.BackgroundTransparency = 0
		DropdownButton.TextColor3 = Colors.Accent
		DropdownButton.TextSize = 11
		DropdownButton.Font = Enum.Font.Gotham
		DropdownButton.Size = UDim2.new(0, 100, 0, 30)
		DropdownButton.Position = UDim2.new(1, -110, 0.5, -15)
		DropdownButton.BorderSizePixel = 0
		DropdownButton.Text = "Select"

		local DropdownButtonCorner = Instance.new("UICorner")
		DropdownButtonCorner.CornerRadius = UDim.new(0, 5)
		DropdownButtonCorner.Parent = DropdownButton

		local CurrentValue = nil
		local Values = Options.Values or {}

		DropdownButton.MouseButton1Click:Connect(function()
			local Index = 1
			for i, v in ipairs(Values) do
				if v == CurrentValue then
					Index = i
					break
				end
			end
			local NextIndex = (Index % #Values) + 1
			CurrentValue = Values[NextIndex]
			DropdownButton.Text = tostring(CurrentValue)
			if Options.Callback then
				Options.Callback(CurrentValue)
			end
		end)

		return {
			GetValue = function()
				return CurrentValue
			end,
			SetValue = function(self, Value)
				CurrentValue = Value
				DropdownButton.Text = tostring(Value)
			end
		}
	end

-- ==========================================
-- SEARCH FUNCTIONALITY
-- ==========================================
SearchInput.Changed:Connect(function()
	local SearchText = SearchInput.Text:lower()
	
	for _, Tab in pairs(VNDHubUI.Tabs) do
		if Tab.Button then
			local TabName = Tab.Button.Text:lower()
			if SearchText == "" or TabName:find(SearchText, 1, true) then
				Tab.Button.Visible = true
			else
				Tab.Button.Visible = false
			end
		end
	end
end)
