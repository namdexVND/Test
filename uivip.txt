local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")
local UserInputService = game:GetService("UserInputService")
local CoreGui = game:GetService("CoreGui")
local LocalPlayer = Players.LocalPlayer

-- Xác định Parent an toàn
local TargetParent = CoreGui
if not pcall(function() return CoreGui.Name end) then
    TargetParent = LocalPlayer:WaitForChild("PlayerGui")
end

local Library = {}

-- Theme mặc định
local Theme = {
    MainBg = Color3.fromRGB(15, 15, 15),
    SidebarBg = Color3.fromRGB(22, 22, 22),
    CardBg = Color3.fromRGB(25, 25, 25),
    AccentRed = Color3.fromRGB(200, 40, 40),
    TextWhite = Color3.fromRGB(255, 255, 255),
    TextGray = Color3.fromRGB(180, 180, 180),
    BorderColor = Color3.fromRGB(45, 45, 45)
}

-- [Hàm Kéo Thả]
local function MakeDraggable(guiObj)
    local dragging, dragInput, dragStart, startPos
    guiObj.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = guiObj.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    guiObj.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            TweenService:Create(guiObj, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            }):Play()
        end
    end)
end

-- [Hàm Khởi Tạo Cửa Sổ]
function Library:CreateWindow(hubTitle)
    local Window = { Tabs = {}, ActiveTab = nil }
    
    local ScreenGui = Instance.new("ScreenGui")
    ScreenGui.Name = "UI_Library"
    ScreenGui.Parent = TargetParent
    ScreenGui.ResetOnSpawn = false

    -- Toggle Button
    local ToggleBtn = Instance.new("ImageButton")
    ToggleBtn.Size = UDim2.new(0, 55, 0, 55)
    ToggleBtn.Position = UDim2.new(0, 15, 0.5, -27)
    ToggleBtn.BackgroundColor3 = Theme.SidebarBg
    ToggleBtn.BorderSizePixel = 0
    ToggleBtn.Image = "rbxassetid://104181747838868"
    ToggleBtn.Parent = ScreenGui
    MakeDraggable(ToggleBtn)
    
    local ToggleCorner = Instance.new("UICorner", ToggleBtn)
    ToggleCorner.CornerRadius = UDim.new(1, 0)
    local ToggleStroke = Instance.new("UIStroke", ToggleBtn)
    ToggleStroke.Color = Theme.AccentRed
    ToggleStroke.Thickness = 1.5

    -- Main Frame
    local MainFrame = Instance.new("Frame")
    MainFrame.Size = UDim2.new(0, 680, 0, 380)
    MainFrame.Position = UDim2.new(0.5, -340, 0.5, -190)
    MainFrame.BackgroundColor3 = Theme.MainBg
    MainFrame.BorderSizePixel = 0
    MainFrame.Visible = true
    MainFrame.Parent = ScreenGui
    MakeDraggable(MainFrame)
    Instance.new("UICorner", MainFrame).CornerRadius = UDim.new(0, 8)
    
    ToggleBtn.MouseButton1Click:Connect(function() MainFrame.Visible = not MainFrame.Visible end)

    local Sidebar = Instance.new("Frame", MainFrame)
    Sidebar.Size = UDim2.new(0, 190, 1, 0)
    Sidebar.BackgroundColor3 = Theme.SidebarBg
    Instance.new("UICorner", Sidebar).CornerRadius = UDim.new(0, 8)

    local TabScroll = Instance.new("ScrollingFrame", Sidebar)
    TabScroll.Size = UDim2.new(1, -10, 1, -65)
    TabScroll.Position = UDim2.new(0, 10, 0, 55)
    TabScroll.BackgroundTransparency = 1
    TabScroll.ScrollBarThickness = 2
    Instance.new("UIListLayout", TabScroll).Padding = UDim.new(0, 5)

    local Container = Instance.new("Frame", MainFrame)
    Container.Size = UDim2.new(1, -200, 1, 0)
    Container.Position = UDim2.new(0, 200, 0, 0)
    Container.BackgroundTransparency = 1
    
    local TabPages = Instance.new("Frame", Container)
    TabPages.Size = UDim2.new(1, -15, 1, -60)
    TabPages.Position = UDim2.new(0, 0, 0, 50)
    TabPages.BackgroundTransparency = 1

    -- [Hàm Tạo Tab]
    function Window:CreateTab(tabName)
        local Tab = { Page = nil, Button = nil, Indicator = nil }
        
        local TabBtnFrame = Instance.new("Frame", TabScroll)
        TabBtnFrame.Size = UDim2.new(1, -10, 0, 35)
        TabBtnFrame.BackgroundTransparency = 1
        
        local Indicator = Instance.new("Frame", TabBtnFrame)
        Indicator.Size = UDim2.new(0, 3, 0, 18)
        Indicator.Position = UDim2.new(0, 2, 0.5, -9)
        Indicator.BackgroundColor3 = Theme.AccentRed
        Indicator.Visible = false
        
        local TabBtn = Instance.new("TextButton", TabBtnFrame)
        TabBtn.Size = UDim2.new(1, -15, 1, 0)
        TabBtn.Position = UDim2.new(0, 12, 0, 0)
        TabBtn.BackgroundTransparency = 1
        TabBtn.Text = tabName
        TabBtn.TextColor3 = Theme.TextGray
        TabBtn.TextXAlignment = Enum.TextXAlignment.Left

        local TabPage = Instance.new("ScrollingFrame", TabPages)
        TabPage.Size = UDim2.new(1, 0, 1, 0)
        TabPage.BackgroundTransparency = 1
        TabPage.ScrollBarThickness = 3
        TabPage.AutomaticCanvasSize = Enum.AutomaticSize.Y
        TabPage.Visible = false
        Instance.new("UIListLayout", TabPage).Padding = UDim.new(0, 8)

        -- Lưu tham chiếu vào Tab
        Tab.Page = TabPage
        Tab.Button = TabBtn
        Tab.Indicator = Indicator

        local function Select()
            if Window.ActiveTab then
                Window.ActiveTab.Page.Visible = false
                Window.ActiveTab.Indicator.Visible = false
                Window.ActiveTab.Button.TextColor3 = Theme.TextGray
            end
            Window.ActiveTab = Tab
            TabPage.Visible = true
            Indicator.Visible = true
            TabBtn.TextColor3 = Theme.TextWhite
        end
        TabBtn.MouseButton1Click:Connect(Select)
        if not Window.ActiveTab then Select() end

        -- [Hàm Tạo Section]
        function Tab:CreateSection(sectionName)
            local Frame = Instance.new("Frame", TabPage)
            Frame.Size = UDim2.new(1, 0, 0, 30)
            Frame.BackgroundTransparency = 1
            local Label = Instance.new("TextLabel", Frame)
            Label.Size = UDim2.new(1, 0, 0, 20)
            Label.Text = sectionName
            Label.TextColor3 = Theme.TextWhite
        end

        -- [Hàm Tạo Button]
        function Tab:CreateButton(btnText, callback)
            local Row = Instance.new("Frame", TabPage)
            Row.Size = UDim2.new(1, 0, 0, 45)
            Row.BackgroundColor3 = Theme.CardBg
            Instance.new("UICorner", Row).CornerRadius = UDim.new(0, 6)
            
            local Btn = Instance.new("TextButton", Row)
            Btn.Size = UDim2.new(0, 110, 0, 30)
            Btn.Position = UDim2.new(1, -122, 0.5, -15)
            Btn.Text = btnText -- Đã sửa: dùng tên biến btnText thay vì hardcode
            Btn.MouseButton1Click:Connect(function() pcall(callback or function() end) end)
        end

        -- [Hàm Tạo Label]
        function Tab:CreateLabel(lblText)
            local Row = Instance.new("Frame", TabPage)
            Row.Size = UDim2.new(1, 0, 0, 50)
            Row.BackgroundColor3 = Theme.CardBg
            Instance.new("UICorner", Row).CornerRadius = UDim.new(0, 6)
            local Lbl = Instance.new("TextLabel", Row)
            Lbl.Size = UDim2.new(1, -24, 1, 0)
            Lbl.Text = lblText
            Lbl.TextColor3 = Theme.TextGray
        end
        return Tab
    end
    return Window
end

return Library
