local tabs = {}
local activeTab = nil

local function switchTab(tabName)
    for name, data in pairs(tabs) do
        local visible = (name == tabName)
        if data.Frame then data.Frame.Visible = visible end
        if data.Button then
            data.Button.BackgroundColor3 = visible and UIColor.Accent or UIColor.Background2
            data.Button.BackgroundTransparency = visible and 0.1 or 0.3
        end
    end
    activeTab = tabName
end

local function CreateWindow(settings)
    local window = {}
    window.Title = settings.Title or "Hub"
    window.SubTitle = settings.SubTitle or ""
    window.Size = settings.Size or UDim2.fromOffset(600, 450)
    window.MinimizeKey = settings.MinimizeKey or Enum.KeyCode.End

    mainFrame.Position = UDim2.new(0.5, -window.Size.X.Offset/2, 0.5, -window.Size.Y.Offset/2)
    titleLabel.Text = window.Title

    -- Phím tắt mở/tắt
    UserInputService.InputBegan:Connect(function(input, processed)
        if processed then return end
        if input.KeyCode == window.MinimizeKey then
            mainGui.Enabled = not mainGui.Enabled
            getgenv().UIToggled = mainGui.Enabled
        end
    end)

                    state = not state
                    toggleBtn.BackgroundColor3 = state and UIColor.ToggleChecked or UIColor.Button
                    toggleBtn.Text = state and "ON" or "OFF"
                    if callback then callback(state) end
                end)

                local obj = {}
                function obj:OnChanged(fn)
                    local oldCb = callback
                    callback = function(v)
                        if oldCb then oldCb(v) end
                        if fn then fn(v) end
                    end
                end
                function obj:SetStage(v)
                    state = v
                    toggleBtn.BackgroundColor3 = v and UIColor.ToggleChecked or UIColor.Button
                    toggleBtn.Text = v and "ON" or "OFF"
                end
                function obj:GetValue() return state end
                function obj:SetValue(v) obj:SetStage(v) end
                return obj
            end

                sliderBox.Size = UDim2.new(0.8, 0, 0, 24)
                sliderBox.Position = UDim2.new(0.1, 0, 0, 22)
                sliderBox.BackgroundColor3 = UIColor.Background2
                sliderBox.Text = tostring(default)
                sliderBox.TextColor3 = UIColor.Text
                sliderBox.Font = Enum.Font.Gotham
                sliderBox.TextSize = 14
                sliderBox.BorderSizePixel = 0
                Instance.new("UICorner", sliderBox).CornerRadius = UDim.new(0, 6)

                local value = default
                sliderBox.FocusLost:Connect(function()
                    local num = tonumber(sliderBox.Text)
                    if num then
                        value = math.clamp(num, min, max)
                        sliderBox.Text = tostring(value)
                        lbl.Text = label .. ": " .. tostring(value)
                        if callback then callback(value) end
                    else
                        sliderBox.Text = tostring(value)
                    end
                end)

                updateHeight()
                local obj = {}
                function obj:OnChanged(fn)
                    local oldCb = callback
                    callback = function(v)
                        if oldCb then oldCb(v) end
                        if fn then fn(v) end
                    end
                end
                function obj:SetValue(v)
                    value = v
                    sliderBox.Text = tostring(v)
                    lbl.Text = label .. ": " .. tostring(v)
                    if callback then callback(v) end
                end
                function obj:GetValue() return value end
                return obj
            end

            -- ===== ADD INPUT =====
            function group:AddInput(id, settings)
                local label = settings.Title or id
                local default = settings.Default or ""
                local placeholder = settings.Placeholder or ""
                local numeric = settings.Numeric or false
                local finished = settings.Finished or false
                local callback = settings.Callback or function() end

                local frame = Instance.new("Frame", content)
                frame.Size = UDim2.new(1, 0, 0, 38)
                frame.BackgroundTransparency = 1

                local lbl = Instance.new("TextLabel", frame)
                lbl.Size = UDim2.new(0.4, -5, 1, 0)
                lbl.BackgroundTransparency = 1
                lbl.Text = label
                lbl.TextColor3 = UIColor.Text
                lbl.Font = Enum.Font.Gotham
                lbl.TextSize = 14
                lbl.TextXAlignment = Enum.TextXAlignment.Left

                local inputBox = Instance.new("TextBox", frame)
                inputBox.Size = UDim2.new(0.5, 0, 0, 28)
                inputBox.Position = UDim2.new(0.45, 0, 0.5, -14)
                inputBox.BackgroundColor3 = UIColor.Background2
                inputBox.Text = default
                inputBox.TextColor3 = UIColor.Text
                inputBox.Font = Enum.Font.Gotham
                inputBox.TextSize = 13
                inputBox.BorderSizePixel = 0
                inputBox.PlaceholderText = placeholder
                Instance.new("UICorner", inputBox).CornerRadius = UDim.new(0, 6)

                if finished then
                    inputBox.FocusLost:Connect(function(enterPressed)
                        if enterPressed then
                            if numeric then
                                local num = tonumber(inputBox.Text)
                                if num then callback(num) end
                            else
                                callback(inputBox.Text)
                            end
                        end
                    end)
                else
                    inputBox:GetPropertyChangedSignal("Text"):Connect(function()
                        if numeric then
                            local num = tonumber(inputBox.Text)
                            if num then callback(num) end
                        else
                            callback(inputBox.Text)
                        end
                    end)
                end

                updateHeight()
                local obj = {}
                function obj:OnChanged(fn)
                    local oldCb = callback
                    callback = function(v)
                        if oldCb then oldCb(v) end
                        if fn then fn(v) end
                    end
                end
                function obj:SetValue(v)
                    inputBox.Text = tostring(v)
                    if callback then callback(v) end
                end
                function obj:GetValue() return inputBox.Text end
                return obj
            end

            -- ===== ADD LABEL (cho Paragraph) =====
            function group:AddLabel(text)
                local lbl = Instance.new("TextLabel", content)
                lbl.Size = UDim2.new(1, 0, 0, 24)
                lbl.BackgroundTransparency = 1
                lbl.Text = text or ""
                lbl.TextColor3 = UIColor.Text
                lbl.Font = Enum.Font.Gotham
                lbl.TextSize = 14
                lbl.TextWrapped = true
                lbl.TextXAlignment = Enum.TextXAlignment.Left
                updateHeight()

                local obj = {}
                function obj:SetText(t) lbl.Text = t end
                function obj:SetDesc(t) lbl.Text = t end
                return obj
            end

            -- ===== ADD PARAGRAPH =====
            function group:AddParagraph(settings)
                local title = settings.Title or settings["Title"] or ""
                local desc = settings.Description or settings["Description"] or settings.Desc or ""
                local fullText = desc ~= "" and (title .. "\n" .. desc) or title
                return group:AddLabel(fullText)
            end

            return group
        end

        -- Tab object
        local tabObj = {}
        function tabObj:AddLeftGroupbox(title)
            return addGroupbox("left", title)
        end
        function tabObj:AddRightGroupbox(title)
            return addGroupbox("right", title)
        end
        local leftRightToggle = true
        function tabObj:AddSection(title)
            if leftRightToggle then
                local g = self:AddLeftGroupbox(title)
                leftRightToggle = false
                return g
            else
                local g = self:AddRightGroupbox(title)
                leftRightToggle = true
                return g
            end
        end
        return tabObj
    end

    return window
end

local function Notify(settings)
    local title = settings.Title or ""
    local desc = settings.Description or ""
    local duration = settings.Duration or 3
    local text = desc ~= "" and (title .. "\n" .. desc) or title

    local frame = Instance.new("Frame", notiContainer)
    frame.Size = UDim2.new(1, 0, 0, 50)
    frame.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
    frame.BackgroundTransparency = 0.2
    Instance.new("UICorner", frame).CornerRadius = UDim.new(0, 8)

    local lbl = Instance.new("TextLabel", frame)
    lbl.Size = UDim2.new(1, -10, 1, 0)
    lbl.Position = UDim2.new(0, 5, 0, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = text
    lbl.TextColor3 = Color3.fromRGB(255, 255, 255)
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 14
    lbl.TextXAlignment = Enum.TextXAlignment.Left

    task.wait(0.1)
    frame.Visible = true

    task.wait(duration)
    TweenService:Create(frame, TweenInfo.new(0.3), {BackgroundTransparency = 1}):Play()
    task.wait(0.3)
    frame:Destroy()
end
