UILibrary.Tabs = {}
local CurrentTab = nil

local function UpdateWindowStyle(Options)
    if Options and Options["Title"] then
        TitleLabel.Text = Options["Title"]
    end
    
    if Options and Options["Desc"] then
        DescLabel.Text = Options["Desc"]
    end
    
    if Options and Options["Image"] then
        LogoImage.Image = Options["Image"]
    end
end

local function CreateTab(TabName)
    -- Tab Button
    local TabButton = Instance.new("TextButton")
    TabButton.Name = TabName
    TabButton.Parent = TabMenuFrame
    TabButton.BackgroundColor3 = UIColors.Element
    TabButton.BackgroundTransparency = 1
    TabButton.TextColor3 = UIColors.Text
    TabButton.TextSize = 11
    TabButton.Font = Enum.Font.Gotham
    TabButton.Size = UDim2.new(1, 0, 0, 40)
    TabButton.Text = TabName
    TabButton.BorderSizePixel = 0
    TabButton.TextWrapped = true

    -- Tab Content
    local TabContent = Instance.new("ScrollingFrame")
    TabContent.Name = TabName .. "Content"
    TabContent.Parent = TabContentFrame
    TabContent.BackgroundTransparency = 1
    TabContent.BorderSizePixel = 0
    TabContent.Size = UDim2.new(1, 0, 1, 0)
    TabContent.ScrollBarThickness = 8
    TabContent.ScrollBarImageColor3 = UIColors.Accent
    TabContent.Visible = false

    local TabContentList = Instance.new("UIListLayout")
    TabContentList.Parent = TabContent
    TabContentList.SortOrder = Enum.SortOrder.LayoutOrder
    TabContentList.Padding = UDim.new(0, 5)

    -- Tab Object
    local Tab = {}
    Tab.Name = TabName
    Tab.Frame = TabContent
    Tab.Button = TabButton
    Tab.Elements = {}

    -- Click event
    TabButton.MouseButton1Click:Connect(function()
        for _, v in pairs(UILibrary.Tabs) do
            v.Frame.Visible = false
            v.Button.BackgroundTransparency = 1
        end
        TabContent.Visible = true
        TabButton.BackgroundTransparency = 0
        TabButton.BackgroundColor3 = UIColors.Accent
        CurrentTab = Tab
    end)

    -- Add methods
    function Tab:AddSection(Name)
        local Section = Instance.new("Frame")
        Section.Name = Name
        Section.Parent = TabContent
        Section.BackgroundTransparency = 1
        Section.Size = UDim2.new(1, 0, 0, 30)

        local SectionTitle = Instance.new("TextLabel")
        SectionTitle.Parent = Section
        SectionTitle.BackgroundTransparency = 1
        SectionTitle.TextColor3 = Color3.fromRGB(255, 255, 255)
        SectionTitle.TextSize = 12
        SectionTitle.Font = Enum.Font.GothamBold
        SectionTitle.Size = UDim2.new(1, -10, 0, 20)
        SectionTitle.Position = UDim2.new(0, 10, 0, 5)
        SectionTitle.TextXAlignment = Enum.TextXAlignment.Left
        SectionTitle.Text = Name

        local Divider = Instance.new("Frame")
        Divider.Parent = Section
        Divider.BackgroundColor3 = UIColors.Accent
        Divider.BackgroundTransparency = 0
        Divider.BorderSizePixel = 0
        Divider.Size = UDim2.new(0.9, 0, 0, 1)
        Divider.Position = UDim2.new(0.05, 0, 0, 25)

        return Section
    end

    function Tab:AddToggle(Identifier, Options)
        local ToggleContainer = Instance.new("Frame")
        ToggleContainer.Name = Identifier
        ToggleContainer.Parent = TabContent
        ToggleContainer.BackgroundTransparency = 1
        ToggleContainer.Size = UDim2.new(1, 0, 0, 40)

        local ToggleLabel = Instance.new("TextLabel")
        ToggleLabel.Parent = ToggleContainer
        ToggleLabel.BackgroundTransparency = 1
        ToggleLabel.TextColor3 = UIColors.Text
        ToggleLabel.TextSize = 11
        ToggleLabel.Font = Enum.Font.Gotham
        ToggleLabel.Size = UDim2.new(0.7, 0, 1, 0)
        ToggleLabel.Position = UDim2.new(0, 10, 0, 0)
        ToggleLabel.TextXAlignment = Enum.TextXAlignment.Left
        ToggleLabel.Text = Options["Title"] or "Toggle"

        local ToggleButton = Instance.new("TextButton")
        ToggleButton.Name = "ToggleButton"
        ToggleButton.Parent = ToggleContainer
        ToggleButton.BackgroundColor3 = UIColors.Element
        ToggleButton.BackgroundTransparency = 0
        ToggleButton.TextColor3 = Color3.fromRGB(255, 255, 255)
        ToggleButton.TextSize = 10
        ToggleButton.Font = Enum.Font.Gotham
        ToggleButton.Size = UDim2.new(0, 40, 0, 20)
        ToggleButton.Position = UDim2.new(1, -55, 0.5, -10)
        ToggleButton.BorderSizePixel = 0
        ToggleButton.Text = "OFF"

        local ToggleValue = Options["Default"] or false
        local OnChanged = nil

        local function UpdateToggle(newValue)
            ToggleValue = newValue
            ToggleButton.BackgroundColor3 = ToggleValue and UIColors.Accent or UIColors.Element
            ToggleButton.Text = ToggleValue and "ON" or "OFF"
            if OnChanged then
                pcall(OnChanged, ToggleValue)
            end
        end

        ToggleButton.MouseButton1Click:Connect(function()
            UpdateToggle(not ToggleValue)
        end)

        local ToggleObject = {}
        function ToggleObject:OnChanged(Callback)
            OnChanged = Callback
            return ToggleObject
        end

        UpdateToggle(ToggleValue)
        return ToggleObject
    end

    function Tab:AddDropdown(Identifier, Options)
        local DropdownContainer = Instance.new("Frame")
        DropdownContainer.Name = Identifier
        DropdownContainer.Parent = TabContent
        DropdownContainer.BackgroundTransparency = 1
        DropdownContainer.Size = UDim2.new(1, 0, 0, 45)

        DropdownLabel.TextSize = 11
        DropdownLabel.Font = Enum.Font.Gotham
        DropdownLabel.Size = UDim2.new(1, 0, 0, 15)
        DropdownLabel.Position = UDim2.new(0, 10, 0, 0)
        DropdownLabel.TextXAlignment = Enum.TextXAlignment.Left
        DropdownLabel.Text = Options["Title"] or "Dropdown"

        local DropdownButton = Instance.new("TextButton")
        DropdownButton.Parent = DropdownContainer
        DropdownButton.BackgroundColor3 = UIColors.Element
        DropdownButton.BackgroundTransparency = 0
        DropdownButton.TextColor3 = UIColors.Text
        DropdownButton.TextSize = 10
        DropdownButton.Font = Enum.Font.Gotham
        DropdownButton.Size = UDim2.new(0.9, 0, 0, 25)
        DropdownButton.Position = UDim2.new(0.05, 0, 0, 18)
        DropdownButton.BorderSizePixel = 0
        DropdownButton.Text = Options["Default"] or "Select"

        local DropdownMenu = Instance.new("Frame")
        DropdownMenu.Name = "Menu"
        DropdownMenu.Parent = DropdownButton
        DropdownMenu.BackgroundColor3 = UIColors.Element
        DropdownMenu.BackgroundTransparency = 0
        DropdownMenu.BorderSizePixel = 0
        DropdownMenu.Size = UDim2.new(1, 0, 0, 0)
        DropdownMenu.Position = UDim2.new(0, 0, 1, 0)
        DropdownMenu.ClipsDescendants = true
        DropdownMenu.Visible = false

        local MenuList = Instance.new("UIListLayout")
        MenuList.Parent = DropdownMenu
        MenuList.SortOrder = Enum.SortOrder.LayoutOrder
        MenuList.Padding = UDim.new(0, 2)

        local OnChanged = nil
        local DropdownOpen = false

        DropdownButton.MouseButton1Click:Connect(function()
            DropdownOpen = not DropdownOpen
            if DropdownOpen then
                local MenuSize = #(Options["Values"] or {}) * 25
                local Tween = TweenService:Create(DropdownMenu, TweenInfo.new(0.2), {Size = UDim2.new(1, 0, 0, MenuSize)})
                Tween:Play()
                DropdownMenu.Visible = true
            else
                local Tween = TweenService:Create(DropdownMenu, TweenInfo.new(0.2), {Size = UDim2.new(1, 0, 0, 0)})
                Tween:Play()
                game:GetService("Debris"):AddItem(DropdownMenu, 0.25)
                wait(0.25)
                DropdownMenu.Visible = false
            end
        end)

        for _, value in ipairs(Options["Values"] or {}) do
            local MenuItem = Instance.new("TextButton")
            MenuItem.Parent = DropdownMenu
            MenuItem.BackgroundColor3 = UIColors.Element
            MenuItem.BackgroundTransparency = 0
            MenuItem.TextColor3 = UIColors.Text
            MenuItem.TextSize = 10
            MenuItem.Font = Enum.Font.Gotham
            MenuItem.Size = UDim2.new(1, 0, 0, 25)
            MenuItem.BorderSizePixel = 0
            MenuItem.Text = value

            MenuItem.MouseButton1Click:Connect(function()
                DropdownButton.Text = value
                DropdownOpen = false
                local Tween = TweenService:Create(DropdownMenu, TweenInfo.new(0.2), {Size = UDim2.new(1, 0, 0, 0)})
                Tween:Play()
                game:GetService("Debris"):AddItem(DropdownMenu, 0.25)
                wait(0.25)
                DropdownMenu.Visible = false
                if OnChanged then
                    pcall(OnChanged, value)
                end
            end)
        end

        local DropdownObject = {}
        function DropdownObject:OnChanged(Callback)
            OnChanged = Callback
            return DropdownObject
        end

        return DropdownObject
    end

    end

    function Tab:AddSlider(Identifier, Options)
        local SliderContainer = Instance.new("Frame")
        SliderContainer.Name = Identifier
        SliderContainer.Parent = TabContent
        SliderContainer.BackgroundTransparency = 1
        SliderContainer.Size = UDim2.new(1, 0, 0, 50)

        local SliderLabel = Instance.new("TextLabel")
        SliderLabel.Parent = SliderContainer
        SliderLabel.BackgroundTransparency = 1
        SliderLabel.TextColor3 = UIColors.Text
        SliderLabel.TextSize = 11
        SliderLabel.Font = Enum.Font.Gotham
        SliderLabel.Size = UDim2.new(1, 0, 0, 15)
        SliderLabel.Position = UDim2.new(0, 10, 0, 0)
        SliderLabel.TextXAlignment = Enum.TextXAlignment.Left
        SliderLabel.Text = Options["Title"] or "Slider"

        local SliderBackground = Instance.new("Frame")
        SliderBackground.Parent = SliderContainer
        SliderBackground.BackgroundColor3 = UIColors.Element
        SliderBackground.BackgroundTransparency = 0
        SliderBackground.BorderSizePixel = 0
        SliderBackground.Size = UDim2.new(0.9, 0, 0, 8)
        SliderBackground.Position = UDim2.new(0.05, 0, 0, 25)

        local SliderButton = Instance.new("TextButton")
        SliderButton.Parent = SliderBackground
        SliderButton.BackgroundColor3 = UIColors.Accent
        SliderButton.BackgroundTransparency = 0
        SliderButton.BorderSizePixel = 0
        SliderButton.Size = UDim2.new(0, 12, 0, 12)
        SliderButton.Position = UDim2.new(0, 0, 0.5, -6)
        SliderButton.Text = ""

        local MinValue = Options["Min"] or 0
        local MaxValue = Options["Max"] or 100
        local DefaultValue = Options["Default"] or MinValue
        local OnChanged = nil

        local function UpdateSlider(Position)
            local Size = SliderBackground.AbsoluteSize.X
            local Value = math.clamp((Position / Size) * (MaxValue - MinValue) + MinValue, MinValue, MaxValue)
            SliderButton.Position = UDim2.new(0, (Position - 6), 0.5, -6)
            if OnChanged then
                pcall(OnChanged, math.floor(Value))
            end
        end

        SliderButton.MouseButton1Down:Connect(function()
            local Connection
            Connection = RunService.InputChanged:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseMovement then
                    local MouseX = UserInputService:GetMouseLocation().X
                    local SliderX = SliderBackground.AbsolutePosition.X
                    UpdateSlider(MouseX - SliderX)
                end
            end)
            UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 then
                    Connection:Disconnect()
                end
            end)
        end)

        local SliderObject = {}
        function SliderObject:OnChanged(Callback)
            OnChanged = Callback
            return SliderObject
        end

        return SliderObject
    end

    UILibrary.Tabs[TabName] = Tab
    
    -- Auto-select first tab
    if #UILibrary.Tabs == 1 then
        TabButton:MouseButton1Click()
    end

    return Tab
end

-- Toggle Shortcut (Insert key)

    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.Insert then
        UILibrary:Toggle()
    end
end)
