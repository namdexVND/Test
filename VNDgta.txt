local function createHealthUI(player)
    if player==LocalPlayer then
        return
    end

    local character=player.Character

    if not character then
        return
    end

    local head=character:FindFirstChild("Head")

    if not head then
        return
    end

    if head:FindFirstChild("HealthVIPGui") then
        return
    end

    local billboard=Instance.new("BillboardGui")
    billboard.Name="HealthVIPGui"
    billboard.Adornee=head
    billboard.Size=UDim2.new(0,150,0,25)
    billboard.StudsOffset=Vector3.new(0,2.5,0)
    billboard.AlwaysOnTop=true
    billboard.Parent=head

    local textLabel=Instance.new("TextLabel")
    textLabel.Name="HealthText"
    textLabel.Size=UDim2.new(1,0,1,0)
    textLabel.BackgroundTransparency=1
    textLabel.TextColor3=Color3.fromRGB(0,255,100)
    textLabel.TextStrokeTransparency=0
    textLabel.TextStrokeColor3=Color3.fromRGB(0,0,0)
    textLabel.Font=Enum.Font.SourceSansBold
    textLabel.TextSize=14
    textLabel.Text="HP"
    textLabel.Parent=billboard
end

local function removeHealthUI(player)
    if player.Character then
        local head=player.Character:FindFirstChild("Head")

        if head then
            local gui=head:FindFirstChild("HealthVIPGui")

            if gui then
                gui:Destroy()
            end
        end
    end
end

local function getClosestPlayer()
    local closestDistance=math.huge
    local closestPlayer=nil

    local character=LocalPlayer.Character
    local myHRP=character and character:FindFirstChild("HumanoidRootPart")

    if not myHRP then
        return nil
    end

    for _,player in pairs(Players:GetPlayers()) do
        if player~=LocalPlayer and player.Character then
            local hrp=player.Character:FindFirstChild("HumanoidRootPart")
            local humanoid=player.Character:FindFirstChildOfClass("Humanoid")

            if hrp and humanoid and humanoid.Health>0 then
                local distance=(hrp.Position-myHRP.Position).Magnitude

                if distance<closestDistance then
                    closestDistance=distance
                    closestPlayer=player
                end
            end
        end
    end

    return closestPlayer
end

local function getAimPosition(target)
    if not target or not target.Character then
        return nil
    end

    local character=target.Character
    local hrp=character:FindFirstChild("HumanoidRootPart")
    local head=character:FindFirstChild("Head")
    local upperTorso=
        character:FindFirstChild("UpperTorso") or
        character:FindFirstChild("Torso")

    if not hrp then
        return nil
    end

    local right=hrp.CFrame.RightVector
    local forward=hrp.CFrame.LookVector
    local up=hrp.CFrame.UpVector
    local base=upperTorso and upperTorso.Position or hrp.Position

    if AimPoint=="FrontBody" then
        return base+
            forward*AimDepthOffset+
            right*AimSideOffset+
            up*AimHeightOffset
    elseif AimPoint=="FrontShoulder" then
        return base+
            up*0.35+
            forward*AimDepthOffset+
            right*(0.75+AimSideOffset)+
            up*AimHeightOffset
    elseif AimPoint=="BackNeck" then
        local neckBase=head and head.Position or base

        return neckBase-
            forward*AimDepthOffset+
            right*AimSideOffset+
            up*AimHeightOffset
    elseif AimPoint=="BackShoulder" then
        return base+
            up*0.35-
            forward*AimDepthOffset+
            right*(0.75+AimSideOffset)+
            up*AimHeightOffset
    end

    return base
end

end)

RunService.RenderStepped:Connect(function()
    if HitboxEnabled then
        for _,player in pairs(Players:GetPlayers()) do
            if player~=LocalPlayer and player.Character then
                local hrp=player.Character:FindFirstChild("HumanoidRootPart")
                local humanoid=player.Character:FindFirstChildOfClass("Humanoid")

                if hrp and humanoid and humanoid.Health>0 then
                    hrp.Size=Vector3.new(
                        HitboxSize,
                        HitboxSize,
                        HitboxSize
                    )

                    hrp.Transparency=1
                    hrp.CanCollide=false
                end
            end
        end
    end

    if ShowHealthEnabled then
        for _,player in pairs(Players:GetPlayers()) do
            if player~=LocalPlayer and player.Character then
                local humanoid=player.Character:FindFirstChildOfClass("Humanoid")
                local head=player.Character:FindFirstChild("Head")

                if humanoid and head then
                    createHealthUI(player)

                    local gui=head:FindFirstChild("HealthVIPGui")
                    local textLabel=gui and gui:FindFirstChild("HealthText")

                    if textLabel then
                        local hp=math.floor(humanoid.Health)
                        local maxHp=math.floor(humanoid.MaxHealth)

                        textLabel.Text=
                            player.Name..
                            " ["..
                            hp..
                            "/"..
                            maxHp..
                            "]"

                        local percent=hp/math.max(maxHp,1)

                        if percent>0.5 then
                            textLabel.TextColor3=Color3.fromRGB(0,255,100)
                        elseif percent>0.25 then
                            textLabel.TextColor3=Color3.fromRGB(255,200,0)
                        else
                            textLabel.TextColor3=Color3.fromRGB(255,50,50)
                        end
                    end
                end
            end
        end
    else
        for _,player in pairs(Players:GetPlayers()) do
            removeHealthUI(player)
        end
    end

    if os.clock()-LastTargetUpdate>=TargetUpdateDelay then
        Target=getClosestPlayer()
        LastTargetUpdate=os.clock()
    end

    local character=LocalPlayer.Character
    local myHRP=character and character:FindFirstChild("HumanoidRootPart")
    local myHumanoid=character and character:FindFirstChildOfClass("Humanoid")

    if not myHRP or not myHumanoid then
        return
    end

    if AutoFollowEnabled and Target and Target.Character then
        local enemyHRP=Target.Character:FindFirstChild("HumanoidRootPart")
        local enemyHumanoid=Target.Character:FindFirstChildOfClass("Humanoid")

        if enemyHRP and enemyHumanoid and enemyHumanoid.Health>0 then
            myHumanoid.WalkSpeed=FollowSpeed

            local targetPosition

            if FollowSide=="Back" then
                targetPosition=
                    enemyHRP.Position-
                    enemyHRP.CFrame.LookVector*
                    FollowDistance
            else
                targetPosition=
                    enemyHRP.Position+
                    enemyHRP.CFrame.LookVector*
                    FollowDistance+
                    enemyHRP.CFrame.RightVector*
                    0.8
            end

            local offset=targetPosition-myHRP.Position
            local distance=offset.Magnitude

            if distance>1.5 then
                myHumanoid.AutoRotate=true

                if distance>0.01 then
                    myHumanoid:Move(
                        offset.Unit,
                        false
                    )
                end
            else
                myHumanoid:Move(
                    Vector3.zero,
                    false
                )

                if FollowSide=="Back" and not AimEnabled then
                    local look=enemyHRP.CFrame.LookVector

                    local flatLook=Vector3.new(
                        look.X,
                        0,
                        look.Z
                    )

                    if flatLook.Magnitude>0.01 then
                        myHumanoid.AutoRotate=false

                        myHRP.CFrame=
                            CFrame.lookAt(
                                myHRP.Position,
                                myHRP.Position+
                                flatLook.Unit
                            )
                    end
                end
            end

            if AimEnabled then
                local aimPosition=getAimPosition(Target)

                if aimPosition then
                    local aimDirection=aimPosition-myHRP.Position

                    local flatDirection=Vector3.new(
                        aimDirection.X,
                        0,
                        aimDirection.Z
                    )

                    if flatDirection.Magnitude>0.01 then
                        myHumanoid.AutoRotate=false

                        myHRP.CFrame=
                            CFrame.lookAt(
                                myHRP.Position,
                                myHRP.Position+
                                flatDirection.Unit
                            )
                    end
                end
            end
        else
            Target=nil
            myHumanoid.AutoRotate=true
        end
    elseif AimEnabled and Target and Target.Character then
        local aimPosition=getAimPosition(Target)

        if aimPosition then
            local aimDirection=aimPosition-myHRP.Position

            local flatDirection=Vector3.new(
                aimDirection.X,
                0,
                aimDirection.Z
            )

            if flatDirection.Magnitude>0.01 then
                myHumanoid.AutoRotate=false

                myHRP.CFrame=
                    CFrame.lookAt(
                        myHRP.Position,
                        myHRP.Position+
                        flatDirection.Unit
                    )
            end
        end
    else
        if not AutoFollowEnabled then
            myHumanoid.WalkSpeed=NormalSpeed
            myHumanoid.AutoRotate=true
        end
    end
end)

RunService.Heartbeat:Connect(function()
    if not AutoAttackEnabled then
        return
    end

    if os.clock()-LastAttack<AttackDelay then
        return
    end

    local character=LocalPlayer.Character

    if not character then
        return
    end

    local myHRP=character:FindFirstChild("HumanoidRootPart")

    if not myHRP then
        return
    end

    local target=Target

    if not target or not target.Character then
        return
    end

    local enemyHRP=target.Character:FindFirstChild("HumanoidRootPart")
    local enemyHumanoid=target.Character:FindFirstChildOfClass("Humanoid")

    if not enemyHRP or not enemyHumanoid or enemyHumanoid.Health<=0 then
        return
    end

    local distance=
        (myHRP.Position-enemyHRP.Position).Magnitude

    if distance<=math.max(
        FollowDistance+4,
        (HitboxSize/2)+3
    ) then
        local tool=character:FindFirstChildOfClass("Tool")

        if tool then
            LastAttack=os.clock()

            pcall(function()
                tool:Activate()
            end)
        end
    end
end)
