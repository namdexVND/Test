local function isAliveModel(model)
    if model and model:FindFirstChild("Humanoid") and model.Humanoid.Health > 0 then
        return true
    end
    return false
end

local function getNearestMonster()
    local char = player.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local hrpPos = char.HumanoidRootPart.Position
    local closestMonster = nil
    local maxDistance = math.huge
    local targetFolder = workspace:FindFirstChild("Bosses") or workspace:FindFirstChild("Monsters") or workspace
    
    for _, obj in ipairs(targetFolder:GetChildren()) do
        if obj:IsA("Model") and isAliveModel(obj) and obj ~= char then
            if not Players:GetPlayerFromCharacter(obj) then 
                local mHrp = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("Torso")
                if mHrp then
                    local distance = (hrpPos - mHrp.Position).Magnitude
                    if distance < maxDistance then
                        maxDistance = distance
                        closestMonster = obj
                    end
                end
            end
        end
    end
    return closestMonster
end

local function isAreaOccupied(targetPos)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            if (p.Character.HumanoidRootPart.Position - targetPos).Magnitude <= 7 then
                return true
            end
        end
    end
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("Seat") or v:IsA("VehicleSeat") then
            if (v.Position - targetPos).Magnitude <= 7 and v.Occupant and v.Occupant.Parent ~= player.Character then 
                return true

        end
    end
    return false

local originalMaterials = {}

local function saveOriginalMaterials()
    if next(originalMaterials) == nil then
        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") then
                originalMaterials[v] = {
                    Material = v.Material,
                    Reflectance = v.Reflectance,
                    CastShadow = v.CastShadow
                }
            end
        end
    end
end

local function toggleGraphicReduction(percent, enable)
    saveOriginalMaterials()
    local totalParts = 0
    for _ in pairs(originalMaterials) do totalParts = totalParts + 1 end
    if totalParts == 0 then return end

    local targetCount = math.floor(totalParts * (percent / 100))
    local currentCount = 0

    for part, data in pairs(originalMaterials) do
        if part and part.Parent then
            currentCount = currentCount + 1
            if currentCount <= targetCount then
                if enable then
                    part.Material = Enum.Material.SmoothPlastic
                    part.Reflectance = 0
                    part.CastShadow = false
                else
                    part.Material = data.Material
                    part.Reflectance = data.Reflectance
                    part.CastShadow = data.CastShadow
                end
            end
        end
    end
end

local function SetupArenaCamera(enabled)

    if enabled then

        if cameraConnection then

            cameraConnection:Disconnect()

            cameraConnection = nil

        end

        if cameraPart then

            cameraPart:Destroy()

            cameraPart = nil

        end

        

        cameraPart = Instance.new("Part")

        cameraPart.Name = "ArenaCameraAnchor"

        cameraPart.Size = Vector3.new(1, 1, 1)

        cameraPart.Transparency = 1

        cameraPart.CanCollide = false

        cameraPart.Anchored = true

        cameraPart.Parent = workspace

        

        workspace.CurrentCamera.CameraType = Enum.CameraType.Scriptable

        currentAngle = 0

        

        cameraConnection = RunService.RenderStepped:Connect(function(deltaTime)

            if enabled then

                currentAngle = currentAngle + (OrbitSpeed * deltaTime)

                

                local offsetX = math.cos(currentAngle) * ArenaRadius

                local offsetZ = math.sin(currentAngle) * ArenaRadius

                

                local currentCenter = ArenaCenter

                local liveBoss = getBoss2()

                if liveBoss and liveBoss:FindFirstChild("HumanoidRootPart") then

                    currentCenter = liveBoss.HumanoidRootPart.Position

                end

                

                local cameraPos = Vector3.new(

                    currentCenter.X + offsetX,

                    currentCenter.Y + CameraHeight,

                    currentCenter.Z + offsetZ

                )

                

                cameraPart.CFrame = CFrame.new(cameraPos, currentCenter)

                workspace.CurrentCamera.CFrame = cameraPart.CFrame

                workspace.CurrentCamera.CameraSubject = cameraPart

            end

        end)

    else

        if cameraConnection then

            cameraConnection:Disconnect()

            cameraConnection = nil

        end

        if cameraPart then

            cameraPart:Destroy()

            cameraPart = nil

        end

        workspace.CurrentCamera.CameraType = Enum.CameraType.Custom

        if Player.Character then

            workspace.CurrentCamera.CameraSubject = Player.Character:FindFirstChildOfClass("Humanoid")

        end

    end

end

local function getClosestPlayer()
    local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end

    local closest, shortest = nil, math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player then
            local char = p.Character
            local hrp  = char and char:FindFirstChild("HumanoidRootPart")
            local hum  = char and char:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                local dist = (myRoot.Position - hrp.Position).Magnitude
                if dist < shortest then
                    shortest = dist
                    closest  = p
                end
            end
        end
    end
    return closest
end

RunService.Heartbeat:Connect(function()
    -- Tắt cả nếu không cần
    if not _G.PvpHitboxActive and not _G.PvpHealthActive then
        if next(activeHitboxes)   then clearAllHitboxes()    end
        if next(activeHealthGuis) then clearAllHealthGuis()  end
        return
    end

    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player then
            local pChar = p.Character
            local pHrp  = pChar and pChar:FindFirstChild("HumanoidRootPart")
            local pHead = pChar and pChar:FindFirstChild("Head")
            local pHum  = pChar and pChar:FindFirstChildOfClass("Humanoid")

            -- === HITBOX ===
            if _G.PvpHitboxActive and pHrp and pHum and pHum.Health > 0 then
                if pHrp.Size ~= Vector3.new(5, 5, 5) then
                    pcall(function()
                        pHrp.Size       = Vector3.new(5, 5, 5)
                        pHrp.CanCollide = false
                    end)
                end
                if not activeHitboxes[p.Name] then
                    local box           = Instance.new("BoxHandleAdornment")
                    box.Size            = Vector3.new(5, 6, 5)
                    box.Color3          = Color3.fromRGB(255, 50, 50)
                    box.Transparency    = 0.55
                    box.AlwaysOnTop     = true
                    box.Adornee         = pHrp
                    box.Parent          = ScreenGui
                    activeHitboxes[p.Name] = box
                end
            else
                if activeHitboxes[p.Name] then
                    pcall(function() activeHitboxes[p.Name]:Destroy() end)
                    activeHitboxes[p.Name] = nil
                end
                if pHrp then
                    pcall(function() pHrp.Size = Vector3.new(2, 2, 1) end)
                end
            end

            -- === ESP CARD ===
            if _G.PvpHealthActive and pHead and pHum and pHum.Health > 0 then
                if not activeHealthGuis[p.Name] then
                    createHealthGui(p, pHead)
                end
                updateHealthGui(activeHealthGuis[p.Name], pHum)
            else
                if activeHealthGuis[p.Name] then
                    pcall(function() activeHealthGuis[p.Name]:Destroy() end)
                    activeHealthGuis[p.Name] = nil
                end
            end
        end
    end
end)

local function getAllConfigFiles()
    local list = {}
    if listfiles then
        local files = listfiles(ConfigFolderName)
        for _, fullPath in ipairs(files) do
            local name = fullPath:match("([^/\\]+)%.json$") or fullPath:match("([^/\\]+)$")
            if name then table.insert(list, name) end
        end
    end
    if #list == 0 then table.insert(list, "Chưa có config nào") end
    return list
end

local function saveNewConfigWithName(customName)
    if customName == "" or customName == "Nhập tên cấu hình cần tạo..." then
        Fluent:Notify({Title = "Lỗi", Content = "Vui lòng nhập tên cấu hình hợp lệ trước!", Duration = 3})
        return
    end
    
    local configData = {}
    for uiName, globalVar in pairs(RegisteredToggles) do
         configData[globalVar] = _G[globalVar] or false
    end
    
    local filePath = ConfigFolderName .. "/" .. customName .. ".json"
    writefile(filePath, HttpService:JSONEncode(configData))
    
    _G.SelectedConfigName = customName
    saveGlobalSystemSettings()
    
    if Fluent.Options and Fluent.Options["SelectConfigDropdown"] then
        local newList = getAllConfigFiles()
        Fluent.Options["SelectConfigDropdown"]:SetValues(newList)
        Fluent.Options["SelectConfigDropdown"]:SetValue(customName)
    end
    
    Fluent:Notify({Title = "Hệ thống Config", Content = "Đã lưu cấu hình đơn: " .. customName, Duration = 3})
end

local function loadSelectedConfig(customName)
    if customName == "Chưa có config nào" or customName == "" then return end
    local filePath = ConfigFolderName .. "/" .. customName .. ".json"
    if isfile(filePath) then
        local fileContent = readfile(filePath)
        local success, data = pcall(function() return HttpService:JSONDecode(fileContent) end)
        
        if success and data then
             for uiName, globalVar in pairs(RegisteredToggles) do
                if data[globalVar] ~= nil then
                    _G[globalVar] = data[globalVar]
                    
                    if Fluent.Options and Fluent.Options[uiName] then
                         Fluent.Options[uiName]:SetValue(data[globalVar])
                    end
                    
                    if globalVar == "FPSBoostState" and data[globalVar] == true then activateFPSBoost() end
                    if globalVar == "Reduce20State" then toggleGraphicReduction(20, data[globalVar]) end
                    if globalVar == "Reduce40State" then toggleGraphicReduction(40, data[globalVar]) end
                    if globalVar == "Reduce60State" then toggleGraphicReduction(60, data[globalVar]) end
                    if globalVar == "Reduce100State" then toggleGraphicReduction(100, data[globalVar]) end
                    if globalVar == "ClearEverythingState" and data[globalVar] == true then
                        activateFPSBoost()
                        for _, v in ipairs(workspace:GetDescendants()) do
                            if v:IsA("Decal") or v:IsA("Texture") or v:IsA("Sound") then pcall(function() v:Destroy() end) end
                            if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") then v.Enabled = false end
                            if v:IsA("BasePart") then v.Material = Enum.Material.SmoothPlastic v.Reflectance = 0 v.CastShadow = false end
                        end
                    end
                end
            end
        end
    end
end

local function deleteSelectedConfig(customName)
    if customName == "Chưa có config nào" or customName == "" then return end
    local filePath = ConfigFolderName .. "/" .. customName .. ".json"
    if isfile(filePath) then
        delfile(filePath)
        _G.SelectedConfigName = ""
        saveGlobalSystemSettings()
        
        local newList = getAllConfigFiles()
        if Fluent.Options and Fluent.Options["SelectConfigDropdown"] then
            Fluent.Options["SelectConfigDropdown"]:SetValues(newList)
            Fluent.Options["SelectConfigDropdown"]:SetValue(newList[1] or "Chưa có config nào")
        end
        
        Fluent:Notify({Title = "Hệ thống Config", Content = "Đã xóa file cấu hình đơn: " .. customName, Duration = 3})
    else
        Fluent:Notify({Title = "Lỗi", Content = "Không tìm thấy file để xóa!", Duration = 3})
    end
end

-- =========================================================
-- ESCAPE RICH TEXT
-- =========================================================

local function EscapeRichText(text)

    text = tostring(text or "")

    text = string.gsub(text, "&", "&amp;")
    text = string.gsub(text, "<", "&lt;")
    text = string.gsub(text, ">", "&gt;")
    text = string.gsub(text, '"', "&quot;")

    return text
end

-- =========================================================
-- COLOR TO HEX
-- =========================================================

local function ColorToHex(color)

    local r = math.floor(color.R * 255 + 0.5)
    local g = math.floor(color.G * 255 + 0.5)
    local b = math.floor(color.B * 255 + 0.5)

    return string.format(
        "#%02X%02X%02X",
        r,
        g,
        b
    )

end

-- =========================================================
-- HEX TO COLOR
-- =========================================================

local function HexToColor(hex)

    hex = tostring(hex or "")
    hex = string.gsub(hex, "#", "")

    if #hex ~= 6 then
        return nil
    end

    local r = tonumber(string.sub(hex, 1, 2), 16)
    local g = tonumber(string.sub(hex, 3, 4), 16)
    local b = tonumber(string.sub(hex, 5, 6), 16)

    if not r or not g or not b then
        return nil
    end

    return Color3.fromRGB(r, g, b)

end

-- =========================================================
-- UPDATE SV
-- =========================================================

local function UpdateSV(x, y)

    x = math.clamp(x, 0, 1)
    y = math.clamp(y, 0, 1)

    CurrentSaturation = x
    CurrentValue = 1 - y

    local color = Color3.fromHSV(
        CurrentHue,
        CurrentSaturation,
        CurrentValue
    )

    if SVMarker then

        SVMarker.Position =
            UDim2.new(
                x,
                -5,
                y,
                -5
            )

    end

    SetTargetColor(color)

end

-- =========================================================
-- UPDATE HUE
-- =========================================================

local function UpdateHue(y)

    y = math.clamp(y, 0, 1)

    CurrentHue = 1 - y

    local color = Color3.fromHSV(
        CurrentHue,
        CurrentSaturation,
        CurrentValue
    )

    if HueMarker then

        HueMarker.Position =
            UDim2.new(
                0,
                -3,
                y,
                -3
            )

    end

    SetTargetColor(color)

end

-- =========================================================
-- TÌM RANK UI
-- =========================================================

local function UpdateRankUIReferences()

    local character =
        LocalPlayer.Character

    if not character then
        return false
    end

    local RankUI =
        character:FindFirstChild(
            "RankUI",
            true
        )

    if not RankUI then

        RankUI =
            character:FindFirstChild(
                "BillboardGui",
                true
            )
            or
            character:FindFirstChild(
                "NameTag",
                true
            )

    end

    if not RankUI then

        warn("❌ KHÔNG TÌM THẤY BẢNG TÊN!")

        return false

    end

    local labels = {}

    for _, obj in pairs(
        RankUI:GetDescendants()
    ) do

        if
            obj:IsA("TextLabel")
            and obj.Visible
        then

            table.insert(
                labels,
                obj
            )

        end

    end

    if #labels == 0 then

        warn("❌ KHÔNG CÓ TEXT LABEL!")

        return false

    end

    labelUsername = nil
    labelDisplay = nil
    labelGroupRank = nil

    local playerName =
        string.lower(
            LocalPlayer.Name
        )

    local displayName =
        string.lower(
            LocalPlayer.DisplayName
        )

    for _, label in pairs(labels) do

        local text =
            string.lower(
                label.Text
            )

        local labelName =
            string.lower(
                label.Name
            )

        -- USERNAME
        if not labelUsername then

            if
                string.find(
                    text,
                    "@"
                )
                or text == playerName
                or text == displayName
                or string.find(
                    labelName,
                    "user"
                )
                or string.find(
                    labelName,
                    "name"
                )
                or string.find(
                    labelName,
                    "display"
                )
            then

                labelUsername = label

            end

        end

        -- RANK
        if
            not labelDisplay
            and label ~= labelUsername
        then

            if
                string.find(
                    text,
                    "rank"
                )
                or string.find(
                    text,
                    "danh"
                )
                or string.find(
                    text,
                    "thành viên"
                )
                or string.find(
                    text,
                    "member"
                )
                or string.find(
                    labelName,
                    "rank"
                )
            then

                labelDisplay = label

            end

        end

        -- LEVEL
        if
            not labelGroupRank
            and label ~= labelUsername
            and label ~= labelDisplay
        then

            if
                string.find(
                    text,
                    "lv"
                )
                or string.find(
                    text,
                    "level"
                )
                or string.find(
                    text,
                    "cấp"
                )
                or string.find(
                    labelName,
                    "level"
                )
            then

                labelGroupRank = label

            end

        end

    end

    -- FALLBACK

    if not labelUsername then
        labelUsername = labels[1]
    end

    if not labelDisplay then

        for _, label in pairs(labels) do

            if label ~= labelUsername then

                labelDisplay = label
                break

            end

        end

    end

    if not labelGroupRank then

        for _, label in pairs(labels) do

            if
                label ~= labelUsername
                and label ~= labelDisplay
            then

                labelGroupRank = label
                break

            end

        end

    end

    isInitialized = true

    print("✅ KẾT NỐI RANK UI THÀNH CÔNG!")

    return true

end

local function GetRainbowColor(offset)

    local hue =
        ((tick() / 5) + offset) % 1

    return Color3.fromHSV(
        hue,
        1,
        1
    )

end

-- =========================================================
-- UPDATE LABEL
-- =========================================================

local function UpdateLabels()

    if not isInitialized then
        return
    end

    -- =====================================================
    -- USERNAME
    -- =====================================================

    if
        labelUsername
        and labelUsername.Parent
    then

        if customUsername then

            labelUsername.Text =
                customUsername

        end

        if RainbowEnabled then

            labelUsername.TextColor3 =
                GetRainbowColor(0)

        else

            labelUsername.TextColor3 =
                UsernameColor

        end

    end

    -- =====================================================
    -- RANK
    -- =====================================================

    if
        labelDisplay
        and labelDisplay.Parent
    then

        if customDisplay then

            labelDisplay.Text =
                customDisplay

        end

        if RainbowEnabled then

            labelDisplay.TextColor3 =
                GetRainbowColor(0.25)

        else

            labelDisplay.TextColor3 =
                RankColor

        end

    end

    -- =====================================================
    -- LEVEL + NAME
    -- =====================================================

    if
        labelGroupRank
        and labelGroupRank.Parent
    then

        if customGroupRank then

            local level =
                BoxLevel
                and BoxLevel.Text
                or "1"

            local name =
                BoxName
                and BoxName.Text
                or "Player"

            if level == "" then
                level = "1"
            end

            if name == "" then
                name = "Player"
            end

            if RainbowEnabled then

                local c1 =
                    GetRainbowColor(0.5)

                local c2 =
                    GetRainbowColor(0.7)

                labelGroupRank.RichText =
                    true

                labelGroupRank.Text =
                    '<font color="' ..
                    ColorToHex(c1) ..
                    '">(Lv ' ..
                    EscapeRichText(level) ..
                    ')</font> ' ..

                    '<font color="' ..
                    ColorToHex(c2) ..
                    '">' ..
                    EscapeRichText(name) ..
                    '</font>'

            else

                labelGroupRank.RichText =
                    true

                labelGroupRank.Text =
                    '<font color="' ..
                    ColorToHex(LevelColor) ..
                    '">(Lv ' ..
                    EscapeRichText(level) ..
                    ')</font> ' ..

                    '<font color="' ..
                    ColorToHex(NameColor) ..
                    '">' ..
                    EscapeRichText(name) ..
                    '</font>'

            end

        end

    end

end

                Fluent.Options["BossSpeedInput"]:SetValue(tostring(_G.BossSettings.Mode2.Speed * 1000))
            end
        end
    end

            if _G.SelectedBossMode == "Chế Độ 1" then

        local n = tonumber(Value)

                _G.BossSettings.Mode2.Speed = n / 1000 -- Chia 1000 cho ổn định
            end
        end

Tabs.Main9:AddInput("ConfigNameInput", {
    Title = "Cấu hình mới",
    Default = "", 
    Placeholder = "Nhập chữ liền không dấu rồi ấn ENTER...",
    Numeric = false,
    Finished = true, 
    Callback = function(Value)
        local cleanedValue = Value:gsub("%s+", "")
        if cleanedValue ~= "" then
            NewConfigNameInput = cleanedValue
            Fluent:Notify({Title = "Hệ thống Config", Content = "✍️ Đã ghi nhận tên: " .. cleanedValue, Duration = 2})
        end
    end
})

            saveNewConfigWithName(NewConfigNameInput) 
        end
    end
})

local ConfigDropdown = Tabs.Main9:AddDropdown("SelectConfigDropdown", {
    Title = "Danh sách",
    Values = getAllConfigFiles(),
    Default = "Chưa có config nào",
    Callback = function(Value)
        if Value ~= "Chưa có config nào" and Value ~= "" then
            _G.SelectedConfigName = Value
            saveGlobalSystemSettings()
            if _G.UseConfigSystem then loadSelectedConfig(Value) end
        end
    end
})

Tabs.Main9:AddToggle("UseConfigSystemToggle", {
    Title = "Sử Dụng Config Đang Chọn",
    Default = false,
    Callback = function(Value)
        _G.UseConfigSystem = Value
        saveGlobalSystemSettings()
        if Value and _G.SelectedConfigName ~= "" and _G.SelectedConfigName ~= "Chưa có config nào" then
            loadSelectedConfig(_G.SelectedConfigName)
        end
    end
})

Tabs.Main9:AddButton({
    Title = "Xóa Cấu Hình Đang Chọn",
    Callback = function()
        if _G.SelectedConfigName ~= "" and _G.SelectedConfigName ~= "Chưa có config nào" then
            deleteSelectedConfig(_G.SelectedConfigName)
        else
            Fluent:Notify({Title = "Lỗi", Content = "Hãy chọn một cấu hình từ danh sách trước khi xóa!", Duration = 3})
        end
    end
})

-- 🚀 ĐỒNG BỘ HÓA TỰ ĐỘNG KHI VÀO GAME
task.defer(function()
    while not Fluent.Options or not Fluent.Options["UseConfigSystemToggle"] or not Fluent.Options["SelectConfigDropdown"] do
        task.wait(0.2)
    end
    
    if isfile(GlobalSettingsFile) then
        pcall(function()
            local data = HttpService:JSONDecode(readfile(GlobalSettingsFile))
            if data and data["SelectedConfigName"] then
                _G.SelectedConfigName = data["SelectedConfigName"]
                _G.UseConfigSystem = data["UseConfigSystem"] or false
                
                Fluent.Options["UseConfigSystemToggle"]:SetValue(_G.UseConfigSystem)
                Fluent.Options["SelectConfigDropdown"]:SetValue(_G.SelectedConfigName)
                
                if _G.UseConfigSystem then
                    loadSelectedConfig(_G.SelectedConfigName)
                    Fluent:Notify({Title = "VND💵 System", Content = "Đã tự động nạp & kích hoạt: " .. _G.SelectedConfigName, Duration = 5})
                end
            end
        end)
    end
end)
