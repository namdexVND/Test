local function isAliveModel(model)
    if model and model:FindFirstChild("Humanoid") and model.Humanoid.Health > 0 then
        return true
    end
    return false
end

local function getNearestMonster()
    local char = player.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local hrpPos = char.HumanoidRootPart.Position
    local closestMonster = nil
    local maxDistance = math.huge
    local targetFolder = workspace:FindFirstChild("Bosses") or workspace:FindFirstChild("Monsters") or workspace
    
    for _, obj in ipairs(targetFolder:GetChildren()) do
        if obj:IsA("Model") and isAliveModel(obj) and obj ~= char then
            if not Players:GetPlayerFromCharacter(obj) then 
                local mHrp = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("Torso")
                if mHrp then
                    local distance = (hrpPos - mHrp.Position).Magnitude
                    if distance < maxDistance then
                        maxDistance = distance
                        closestMonster = obj
                    end
                end
            end
        end
    end
    return closestMonster
end

local function isAreaOccupied(targetPos)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            if (p.Character.HumanoidRootPart.Position - targetPos).Magnitude <= 7 then
                return true
            end
        end
    end
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("Seat") or v:IsA("VehicleSeat") then
            if (v.Position - targetPos).Magnitude <= 7 and v.Occupant and v.Occupant.Parent ~= player.Character then 
                return true 
            end
        end
    end
    return false
end

local originalMaterials = {}

local function saveOriginalMaterials()
    if next(originalMaterials) == nil then
        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") then
                originalMaterials[v] = {
                    Material = v.Material,
                    Reflectance = v.Reflectance,
                    CastShadow = v.CastShadow
                }
            end
        end
    end
end

local function toggleGraphicReduction(percent, enable)
    saveOriginalMaterials()
    local totalParts = 0
    for _ in pairs(originalMaterials) do totalParts = totalParts + 1 end
    if totalParts == 0 then return end

    local targetCount = math.floor(totalParts * (percent / 100))
    local currentCount = 0

    for part, data in pairs(originalMaterials) do
        if part and part.Parent then
            currentCount = currentCount + 1
            if currentCount <= targetCount then
                if enable then
                    part.Material = Enum.Material.SmoothPlastic
                    part.Reflectance = 0
                    part.CastShadow = false
                else
                    part.Material = data.Material
                    part.Reflectance = data.Reflectance
                    part.CastShadow = data.CastShadow
                end
            end
        end
    end
end

local NPC2_CENTER = Vector3.new(-2790,239,-1762)
local CurrentPoint = 1
local LastTeleport = 0
local LastClick = 0

local POINT_ANGLES = {
    math.rad(-90),
    math.rad(30),
    math.rad(150)
}

local function FindNPC2()
    local npc2 = workspace:FindFirstChild("NPC2",true)
    if not npc2 then return nil,nil end
    if npc2:IsA("Model") then
        local root = npc2:FindFirstChild("HumanoidRootPart") or npc2.PrimaryPart or npc2:FindFirstChildWhichIsA("BasePart")
        if root then return npc2,root end
    elseif npc2:IsA("BasePart") then
        return npc2,npc2
    end
    return nil,nil
end

local function GetCenterPoint(index)
    local distance = tonumber(_G.BossSettings.Mode3.Dist) or 17
    if distance < 0 then distance = 0 end
    local angle = POINT_ANGLES[index] or POINT_ANGLES[1]
    local offset = Vector3.new(math.cos(angle)*distance,0,math.sin(angle)*distance)
    return Vector3.new(NPC2_CENTER.X+offset.X,NPC2_CENTER.Y,NPC2_CENTER.Z+offset.Z)
end

local function TeleportToPoint(index)
    local Character = player.Character
    if not Character then return end
    local HRP = Character:FindFirstChild("HumanoidRootPart")
    if not HRP then return end

    local Boss,BossRoot = FindNPC2()
    if not BossRoot then return end

    local Target = GetCenterPoint(index)
    local LookAt = Vector3.new(BossRoot.Position.X,Target.Y,BossRoot.Position.Z)

    HRP.CFrame = CFrame.lookAt(Target,LookAt)
    HRP.AssemblyLinearVelocity = Vector3.zero
end

RunService.Heartbeat:Connect(function()
    if _G.SelectedBossMode ~= "Chế Độ 3" or not _G.AutoFarmBossCircle then return end

    local now = os.clock()
    local speed = tonumber(_G.BossSettings.Mode3.Speed) or 0.08
    if speed <= 0 then speed = 0.01 end
    if now - LastTeleport < speed then return end
    LastTeleport = now

    local Character = player.Character
    local HRP = Character and Character:FindFirstChild("HumanoidRootPart")
    if not HRP then return end

    local Boss,BossRoot = FindNPC2()
    if not BossRoot then return end

    TeleportToPoint(CurrentPoint)

    CurrentPoint += 1
    if CurrentPoint > 3 then
        CurrentPoint = 1
    end
end)

local ArenaRadius = 55

local CameraHeight = 60

local OrbitSpeed = 0.3

local currentAngle = 0

local cameraConnection = nil

local cameraPart = nil

local function getBoss2()

    local gh2 = workspace:FindFirstChild("GiangHo2")

    if not gh2 then return nil end

    local npcs = gh2:FindFirstChild("NPCs")

    if not npcs then return nil end

    return npcs:FindFirstChild("NPC2")

local function SetupArenaCamera(enabled)

    if enabled then

        if cameraConnection then

            cameraConnection:Disconnect()

            cameraConnection = nil

        end

        if cameraPart then

            cameraPart:Destroy()

            cameraPart = nil

        end

        

        cameraPart = Instance.new("Part")

        cameraPart.Name = "ArenaCameraAnchor"

        cameraPart.Size = Vector3.new(1, 1, 1)

        cameraPart.Transparency = 1

        cameraPart.CanCollide = false

        cameraPart.Anchored = true

        cameraPart.Parent = workspace

        

        workspace.CurrentCamera.CameraType = Enum.CameraType.Scriptable

        currentAngle = 0

        

        cameraConnection = RunService.RenderStepped:Connect(function(deltaTime)

            if enabled then

                currentAngle = currentAngle + (OrbitSpeed * deltaTime)

                

                local offsetX = math.cos(currentAngle) * ArenaRadius

                local offsetZ = math.sin(currentAngle) * ArenaRadius

                

                local currentCenter = ArenaCenter

                local liveBoss = getBoss2()

                if liveBoss and liveBoss:FindFirstChild("HumanoidRootPart") then

                    currentCenter = liveBoss.HumanoidRootPart.Position

                end

                

                local cameraPos = Vector3.new(

                    currentCenter.X + offsetX,

                    currentCenter.Y + CameraHeight,

                    currentCenter.Z + offsetZ

                )

                

                cameraPart.CFrame = CFrame.new(cameraPos, currentCenter)

                workspace.CurrentCamera.CFrame = cameraPart.CFrame

                workspace.CurrentCamera.CameraSubject = cameraPart

            end

        end)

    else

        if cameraConnection then

            cameraConnection:Disconnect()

            cameraConnection = nil

        end

        if cameraPart then

            cameraPart:Destroy()

            cameraPart = nil

        end

        workspace.CurrentCamera.CameraType = Enum.CameraType.Custom

        if Player.Character then

            workspace.CurrentCamera.CameraSubject = Player.Character:FindFirstChildOfClass("Humanoid")

        end

    end

end

local function getClosestPlayer()
    local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end

    local closest, shortest = nil, math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player then
            local char = p.Character
            local hrp  = char and char:FindFirstChild("HumanoidRootPart")
            local hum  = char and char:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                local dist = (myRoot.Position - hrp.Position).Magnitude
                if dist < shortest then
                    shortest = dist
                    closest  = p
                end
            end
        end
    end
    return closest
end

RunService.RenderStepped:Connect(function()
    pcall(function()
        local char = player.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        local hum  = char and char:FindFirstChildOfClass("Humanoid")

        -- 1. Speed (CFrame – không bị kick)
        if _G.PvpSpeedActive and hrp and hum and hum.MoveDirection.Magnitude > 0 then

            if not hrp:FindFirstChild("VND_SpinVelocity") then

                spin.Name                = "VND_SpinVelocity"
                spin.MaxTorque           = Vector3.new(0, math.huge, 0)
                spin.AngularVelocity     = Vector3.new(0, 40, 0)
                spin.Parent              = hrp
            end
        else
            if hrp then
                local sv = hrp:FindFirstChild("VND_SpinVelocity")
                if sv then sv:Destroy() end
            end
        end

        -- 3. Aim (không di chuyển camera)
        if _G.PvpAimNoCamActive and hrp then
            local target = getClosestPlayer()
            if target and target.Character then
                local tHrp = target.Character:FindFirstChild("HumanoidRootPart")
                if tHrp then
                    local lookAt = Vector3.new(tHrp.Position.X, hrp.Position.Y, tHrp.Position.Z)
                    hrp.CFrame   = CFrame.new(hrp.Position, lookAt)
                end
            end
        end
    end)
end)

RunService.Heartbeat:Connect(function()
    -- Tắt cả nếu không cần
    if not _G.PvpHitboxActive and not _G.PvpHealthActive then
        if next(activeHitboxes)   then clearAllHitboxes()    end
        if next(activeHealthGuis) then clearAllHealthGuis()  end
        return
    end

    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player then
            local pChar = p.Character
            local pHrp  = pChar and pChar:FindFirstChild("HumanoidRootPart")
            local pHead = pChar and pChar:FindFirstChild("Head")
            local pHum  = pChar and pChar:FindFirstChildOfClass("Humanoid")

            -- === HITBOX ===
            if _G.PvpHitboxActive and pHrp and pHum and pHum.Health > 0 then
                if pHrp.Size ~= Vector3.new(5, 5, 5) then
                    pcall(function()
                        pHrp.Size       = Vector3.new(5, 5, 5)
                        pHrp.CanCollide = false
                    end)
                end
                if not activeHitboxes[p.Name] then
                    local box           = Instance.new("BoxHandleAdornment")
                    box.Size            = Vector3.new(5, 6, 5)
                    box.Color3          = Color3.fromRGB(255, 50, 50)
                    box.Transparency    = 0.55
                    box.AlwaysOnTop     = true
                    box.Adornee         = pHrp
                    box.Parent          = ScreenGui
                    activeHitboxes[p.Name] = box
                end
            else
                if activeHitboxes[p.Name] then
                    pcall(function() activeHitboxes[p.Name]:Destroy() end)
                    activeHitboxes[p.Name] = nil
                end
                if pHrp then
                    pcall(function() pHrp.Size = Vector3.new(2, 2, 1) end)
                end
            end

            -- === ESP CARD ===
            if _G.PvpHealthActive and pHead and pHum and pHum.Health > 0 then
                if not activeHealthGuis[p.Name] then
                    createHealthGui(p, pHead)
                end
                updateHealthGui(activeHealthGuis[p.Name], pHum)
            else
                if activeHealthGuis[p.Name] then
                    pcall(function() activeHealthGuis[p.Name]:Destroy() end)
                    activeHealthGuis[p.Name] = nil
                end
            end
        end
    end
end)

local function getAllConfigFiles()
    local list = {}
    if listfiles then
        local files = listfiles(ConfigFolderName)

            local name = fullPath:match("([^/\\]+)%.json$") or fullPath:match("([^/\\]+)$")

        local fileContent = readfile(filePath)
        local success, data = pcall(function() return HttpService:JSONDecode(fileContent) end)
        
        if success and data then
             for uiName, globalVar in pairs(RegisteredToggles) do
                if data[globalVar] ~= nil then
                    _G[globalVar] = data[globalVar]
                    
                    if Fluent.Options and Fluent.Options[uiName] then
                         Fluent.Options[uiName]:SetValue(data[globalVar])
                    end
                    
                    if globalVar == "FPSBoostState" and data[globalVar] == true then activateFPSBoost() end
                    if globalVar == "Reduce20State" then toggleGraphicReduction(20, data[globalVar]) end
                    if globalVar == "Reduce40State" then toggleGraphicReduction(40, data[globalVar]) end
                    if globalVar == "Reduce60State" then toggleGraphicReduction(60, data[globalVar]) end
                    if globalVar == "Reduce100State" then toggleGraphicReduction(100, data[globalVar]) end
                    if globalVar == "ClearEverythingState" and data[globalVar] == true then
                        activateFPSBoost()
                        for _, v in ipairs(workspace:GetDescendants()) do

local function EscapeRichText(text)
	text = tostring(text or "")

	text = string.gsub(text, "&", "&amp;")
	text = string.gsub(text, "<", "&lt;")
	text = string.gsub(text, ">", "&gt;")
	text = string.gsub(text, '"', "&quot;")

	return text
end

local function ColorToHex(color)
	local r = math.floor(color.R * 255 + 0.5)
	local g = math.floor(color.G * 255 + 0.5)
	local b = math.floor(color.B * 255 + 0.5)

	return string.format("#%02X%02X%02X", r, g, b)
end

local function HexToColor(hex)
	hex = tostring(hex or "")

	hex = string.gsub(hex, "#", "")
	hex = string.gsub(hex, "%s+", "")

	if #hex ~= 6 then
		return nil
	end

	local r = tonumber(string.sub(hex,1,2),16)
	local g = tonumber(string.sub(hex,3,4),16)
	local b = tonumber(string.sub(hex,5,6),16)

	if not r or not g or not b then
		return nil
	end

	return Color3.fromRGB(r,g,b)
end

local function UpdateSV(x,y)

	x = math.clamp(x,0,1)
	y = math.clamp(y,0,1)

	CurrentSaturation = x
	CurrentValue = 1-y

	if SVMarker then
		SVMarker.Position = UDim2.new(x,0,y,0)
	end

	SetTargetColor(
		Color3.fromHSV(
			CurrentHue,
			CurrentSaturation,
			CurrentValue
		)
	)
end

local function UpdateHue(y)

	y = math.clamp(y,0,1)

	CurrentHue = 1-y

	if HueMarker then
		HueMarker.Position = UDim2.new(0,-3,y,-3)
	end

	SetTargetColor(
		Color3.fromHSV(
			CurrentHue,
			CurrentSaturation,
			CurrentValue
		)
	)
end

-- CHECK IF OBJECT BELONGS TO LOCAL CHARACTER
-- =========================================================

local function IsInsideLocalCharacter(object)

	if not object then
		return false
	end

	local character = LocalPlayer.Character

	if not character then
		return false
	end

	return object:IsDescendantOf(character)
end

-- =========================================================
-- FIND VALID RANK UI
-- =========================================================

local function FindValidRankUI()

	local character = LocalPlayer.Character

	if not character then
		return nil
	end

	-- Không dùng character cũ
	if LastCharacter ~= character then

		LastCharacter = character
		RankUIReference = nil
		RankUIStableCount = 0

		labelUsername = nil
		labelDisplay = nil
		labelGroupRank = nil

		isInitialized = false
	end

	-- =====================================================
	-- ƯU TIÊN RANKUI CHÍNH XÁC
	-- =====================================================

	local rankUI = character:FindFirstChild("RankUI")

	if rankUI and IsInsideLocalCharacter(rankUI) then
		return rankUI
	end

	-- =====================================================
	-- TÌM BILLBOARDGUI THUỘC LOCAL CHARACTER
	-- KHÔNG LẤY OBJECT NGOÀI CHARACTER
	-- =====================================================

	for _,obj in ipairs(character:GetDescendants()) do

		if obj:IsA("BillboardGui") then

			-- Nếu có Adornee thì phải trỏ vào character
			if obj.Adornee then

				if obj.Adornee:IsDescendantOf(character) then
					return obj
				end

			else
				return obj
			end
		end
	end

	-- NameTag nếu thật sự nằm trong Character
	local nameTag = character:FindFirstChild("NameTag",true)

	if nameTag and IsInsideLocalCharacter(nameTag) then
		return nameTag
	end

	return nil
end

-- =========================================================
-- GET LABELS FROM VALID UI ONLY
-- =========================================================

local function GetValidLabels(rankUI)

	if not rankUI then
		return {}
	end

	if not IsInsideLocalCharacter(rankUI) then
		return {}
	end

	local labels = {}

	for _,obj in ipairs(rankUI:GetDescendants()) do

		if obj:IsA("TextLabel") then

			-- Chỉ nhận label thuộc đúng UI
			if IsInsideLocalCharacter(obj) then
				table.insert(labels,obj)
			end
		end
	end

	return labels
end

-- =========================================================
-- FIND RANK LABELS
-- =========================================================

local function ResolveRankLabels(rankUI)

	local labels = GetValidLabels(rankUI)

	if #labels == 0 then
		return false
	end

	local usernameLabel
	local displayLabel
	local groupRankLabel

	local playerName = string.lower(LocalPlayer.Name)
	local displayName = string.lower(LocalPlayer.DisplayName)

	-- =====================================================
	-- TÌM CHÍNH XÁC
	-- =====================================================

	for _,label in ipairs(labels) do

		local text = string.lower(tostring(label.Text or ""))
		local name = string.lower(tostring(label.Name or ""))

		-- USERNAME

		if not usernameLabel then

			if text == "@"..playerName
				or text == playerName
				or text == displayName
				or string.find(name,"username",1,true)
				or string.find(name,"displayname",1,true) then

				usernameLabel = label
			end
		end

		-- RANK

		if not displayLabel
			and label ~= usernameLabel then

			if string.find(text,"rank",1,true)
				or string.find(text,"danh",1,true)
				or string.find(text,"member",1,true)
				or string.find(name,"rank",1,true) then

				displayLabel = label
			end
		end

		-- LEVEL

		if not groupRankLabel
			and label ~= usernameLabel
			and label ~= displayLabel then

			if string.find(text,"lv",1,true)
				or string.find(text,"level",1,true)
				or string.find(text,"cấp",1,true)
				or string.find(name,"level",1,true) then

				groupRankLabel = label
			end
		end
	end

	-- =====================================================
	-- FALLBACK AN TOÀN
	-- =====================================================

	if not usernameLabel then
		usernameLabel = labels[1]
	end

	if not displayLabel then

		for _,label in ipairs(labels) do

			if label ~= usernameLabel then
				displayLabel = label
				break
			end
		end
	end

	if not groupRankLabel then

		for _,label in ipairs(labels) do

			if label ~= usernameLabel
				and label ~= displayLabel then

				groupRankLabel = label
				break
			end
		end
	end

	-- Phải có ít nhất 3 label mới xác nhận
	-- tránh bắt nhầm UI trong lúc spawn
	if not usernameLabel
		or not displayLabel
		or not groupRankLabel then

		return false
	end

	-- =====================================================
	-- XÁC NHẬN OBJECT VẪN THUỘC CHARACTER
	-- =====================================================

	if not IsInsideLocalCharacter(usernameLabel)
		or not IsInsideLocalCharacter(displayLabel)
		or not IsInsideLocalCharacter(groupRankLabel) then

		return false
	end

	-- =====================================================
	-- SET REFERENCES
	-- =====================================================

	labelUsername = usernameLabel
	labelDisplay = displayLabel
	labelGroupRank = groupRankLabel

	return true
end

-- =========================================================
-- UPDATE RANK UI REFERENCES
-- =========================================================

local function UpdateRankUIReferences()

	local character = LocalPlayer.Character

	if not character then
		return false
	end

	-- Không cho dùng reference của Character cũ
	if LastCharacter ~= character then

		LastCharacter = character

		RankUIReference = nil

		labelUsername = nil
		labelDisplay = nil
		labelGroupRank = nil

		RankUIStableCount = 0
		isInitialized = false
	end

	-- Nếu RankUI cũ không còn thuộc character
	if RankUIReference
		and not IsInsideLocalCharacter(RankUIReference) then

		RankUIReference = nil

		labelUsername = nil
		labelDisplay = nil
		labelGroupRank = nil

		RankUIStableCount = 0
		isInitialized = false
	end

	-- Tìm UI
	local foundUI = FindValidRankUI()

	if not foundUI then

		RankUIStableCount = 0
		isInitialized = false

		return false
	end

	-- UI phải giống nhau qua nhiều lần scan
	if RankUIReference == foundUI then

		RankUIStableCount += 1

	else

		RankUIReference = foundUI
		RankUIStableCount = 1

		labelUsername = nil
		labelDisplay = nil
		labelGroupRank = nil

		isInitialized = false
	end

	-- Chưa đủ ổn định
	if RankUIStableCount < MIN_STABLE_COUNT then
		return false
	end

	-- Resolve labels
	if not ResolveRankLabels(foundUI) then

		isInitialized = false

		return false
	end

	-- =====================================================
	-- CHỈ LƯU MÀU GỐC KHI ĐÃ XÁC NHẬN UI ĐÚNG
	-- =====================================================

	if not colorChanged then
		SaveOriginalColors()
	end

	isInitialized = true

	return true
end

-- =========================================================
-- INITIALIZE
-- =========================================================

local function Initialize()

	isInitialized = false
	RankUIStableCount = 0
	RankUIReference = nil

	-- Không đụng vào label ngay khi script vừa load
	task.wait(2)

	-- Chỉ tìm UI thuộc Character hiện tại
	for i = 1,5 do

		if UpdateRankUIReferences() then
			break
		end

		task.wait(0.5)
	end
end

-- =========================================================

LocalPlayer.CharacterAdded:Connect(function(character)

	-- RESET TOÀN BỘ REFERENCE CŨ

	isInitialized = false

	LastCharacter = character
	RankUIReference = nil

	labelUsername = nil
	labelDisplay = nil
	labelGroupRank = nil

	RankUIStableCount = 0

	-- Không đụng UI trong thời gian character đang spawn
	task.spawn(function()

		task.wait(2)

		for i = 1,8 do

			if UpdateRankUIReferences() then
				break
			end

			task.wait(0.5)
		end
	end)
end)

-- =========================================================
-- APPLY CHANGES
-- =========================================================

local function ApplyChanges()

	if isApplying then
		return
	end

	isApplying = true

	local changed = false

	if BoxUser and BoxUser.Text ~= "" then

		customUsername = BoxUser.Text
		changed = true
	end

	if BoxRank and BoxRank.Text ~= "" then

		customDisplay = BoxRank.Text
		changed = true
	end

	if (BoxLevel and BoxLevel.Text ~= "")
		or (BoxName and BoxName.Text ~= "") then

		local lv =
			(BoxLevel and BoxLevel.Text ~= "")
			and BoxLevel.Text
			or "1"

		local name =
			(BoxName and BoxName.Text ~= "")
			and BoxName.Text
			or "Player"

		customGroupRank = "(Lv "..lv..") "..name

		changed = true
	end

	if colorChanged then
		changed = true
	end

	if RainbowEnabled then
		changed = true
	end

	if changed then

		if StatusBar then

			StatusBar.Text = "●  Đã áp dụng!"
			StatusBar.TextColor3 =
				Color3.fromRGB(80,255,150)
		end

		if ApplyBtn then

			ApplyBtn.Text = "✓  ĐÃ ÁP DỤNG"

			task.wait(0.25)

			ApplyBtn.Text = "✦  ÁP DỤNG"
		end

		if StatusBar then

			StatusBar.Text = "●  Sẵn sàng"
			StatusBar.TextColor3 =
				Color3.fromRGB(180,130,255)
		end

	else

		if StatusBar then

			StatusBar.Text = "⚠  Không có thay đổi"
			StatusBar.TextColor3 =
				Color3.fromRGB(255,190,80)
		end
	end

	isApplying = false
end

-- =========================================================
-- ONE SYNCHRONIZED RAINBOW COLOR
-- =========================================================

local function GetRainbowColor()

	-- TẤT CẢ LABEL DÙNG CHUNG HUE NÀY
	local hue = (os.clock() / 5) % 1

	RainbowColor = Color3.fromHSV(
		hue,
		1,
		1
	)

	return RainbowColor
end

-- =========================================================
-- APPLY SAME COLOR TO ALL LABELS
-- =========================================================

local function ApplyRainbowColor(color)

	if not color then
		return
	end

	-- USERNAME

	if labelUsername
		and labelUsername.Parent
		and IsInsideLocalCharacter(labelUsername) then

		labelUsername.TextColor3 = color
	end

	-- RANK

	if labelDisplay
		and labelDisplay.Parent
		and IsInsideLocalCharacter(labelDisplay) then

		labelDisplay.TextColor3 = color
	end

	-- LEVEL / NAME
	--
	-- RichText cũng dùng CHUNG 1 COLOR
	-- không còn 2 màu riêng.

	if labelGroupRank
		and labelGroupRank.Parent
		and IsInsideLocalCharacter(labelGroupRank) then

		if customGroupRank then

			local level =
				(BoxLevel and BoxLevel.Text ~= "")
				and BoxLevel.Text
				or "1"

			local name =
				(BoxName and BoxName.Text ~= "")
				and BoxName.Text
				or "Player"

			local hex = ColorToHex(color)

			labelGroupRank.RichText = true

			labelGroupRank.Text =
				'<font color="'..hex..'">'..
				'(Lv '..
				EscapeRichText(level)..
				') '..
				EscapeRichText(name)..
				'</font>'

		else

			labelGroupRank.TextColor3 = color
		end
	end
end

			local level =
				(BoxLevel and BoxLevel.Text ~= "")
				and BoxLevel.Text
				or "1"

			local name =
				(BoxName and BoxName.Text ~= "")
				and BoxName.Text
				or "Player"

	local character = LocalPlayer.Character

	if not character then

		isInitialized = false
		return
	end

	if LastCharacter ~= character then

		isInitialized = false
		return
	end

	if not RankUIReference
		or not IsInsideLocalCharacter(RankUIReference) then

		isInitialized = false
		RankUIReference = nil

		return
	end

	if not labelUsername
		or not labelDisplay
		or not labelGroupRank then

		isInitialized = false
		return
	end

	if not IsInsideLocalCharacter(labelUsername)
		or not IsInsideLocalCharacter(labelDisplay)
		or not IsInsideLocalCharacter(labelGroupRank) then

		isInitialized = false

		labelUsername = nil
		labelDisplay = nil
		labelGroupRank = nil

		return
	end

	-- =====================================================
	-- TEXT
	-- =====================================================

	if customUsername then
		labelUsername.Text = customUsername
	end

	if customDisplay then
		labelDisplay.Text = customDisplay
	end

	-- =====================================================
	-- RAINBOW ON
	-- =====================================================

	if RainbowEnabled then

		-- QUAN TRỌNG:
		-- CHỈ TẠO 1 COLOR DUY NHẤT
		--
		-- Không dùng:
		-- GetRainbowColor(0)
		-- GetRainbowColor(0.25)
		-- GetRainbowColor(0.5)
		-- GetRainbowColor(0.7)
		--
		-- nữa.

		local sameRainbowColor = GetRainbowColor()

		ApplyRainbowColor(sameRainbowColor)

		return
	end

	-- =====================================================
	-- RAINBOW OFF
	-- =====================================================

	if colorChanged then

		ApplyCustomColors()

	end
end

-- =========================================================
-- START
-- =========================================================

task.spawn(function()
	Initialize()
end)

-- =========================================================
-- RENDER
-- =========================================================

RunService.RenderStepped:Connect(function()

	if isInitialized then
		UpdateLabels()
	end
end)

	while task.wait(1) do

		local character = LocalPlayer.Character

		if not character then

			isInitialized = false
			continue
		end

		if LastCharacter ~= character then

			isInitialized = false
			RankUIReference = nil

			labelUsername = nil
			labelDisplay = nil
			labelGroupRank = nil

			RankUIStableCount = 0

			continue
		end

		if not isInitialized then

			UpdateRankUIReferences()
		end
	end
end)

    while not Fluent.Options or not Fluent.Options["UseConfigSystemToggle"] or not Fluent.Options["SelectConfigDropdown"] do
        task.wait(0.2)
    end
    
    if isfile(GlobalSettingsFile) then
        pcall(function()
            local data = HttpService:JSONDecode(readfile(GlobalSettingsFile))
            if data and data["SelectedConfigName"] then
                _G.SelectedConfigName = data["SelectedConfigName"]
                _G.UseConfigSystem = data["UseConfigSystem"] or false
                
                Fluent.Options["UseConfigSystemToggle"]:SetValue(_G.UseConfigSystem)
                Fluent.Options["SelectConfigDropdown"]:SetValue(_G.SelectedConfigName)

                end
            end
        end)
    end
