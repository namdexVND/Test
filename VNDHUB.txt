local function isAliveModel(model)
    if model and model:FindFirstChild("Humanoid") and model.Humanoid.Health > 0 then
        return true
    end
    return false
end

local function getNearestMonster()
    local char = player.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local hrpPos = char.HumanoidRootPart.Position
    local closestMonster = nil
    local maxDistance = math.huge
    local targetFolder = workspace:FindFirstChild("Bosses") or workspace:FindFirstChild("Monsters") or workspace
    
    for _, obj in ipairs(targetFolder:GetChildren()) do
        if obj:IsA("Model") and isAliveModel(obj) and obj ~= char then
            if not Players:GetPlayerFromCharacter(obj) then 
                local mHrp = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("Torso")
                if mHrp then
                    local distance = (hrpPos - mHrp.Position).Magnitude
                    if distance < maxDistance then
                        maxDistance = distance
                        closestMonster = obj
                    end
                end
            end
        end
    end
    return closestMonster
end

local function isAreaOccupied(targetPos)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            if (p.Character.HumanoidRootPart.Position - targetPos).Magnitude <= 7 then
                return true
            end
        end
    end
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("Seat") or v:IsA("VehicleSeat") then
            if (v.Position - targetPos).Magnitude <= 7 and v.Occupant and v.Occupant.Parent ~= player.Character then 
                return true 
            end
        end
    end
    return false
end

local originalMaterials = {}

local function saveOriginalMaterials()
    if next(originalMaterials) == nil then
        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") then
                originalMaterials[v] = {
                    Material = v.Material,
                    Reflectance = v.Reflectance,
                    CastShadow = v.CastShadow
                }
            end
        end
    end
end

local function toggleGraphicReduction(percent, enable)
    saveOriginalMaterials()
    local totalParts = 0
    for _ in pairs(originalMaterials) do totalParts = totalParts + 1 end
    if totalParts == 0 then return end

    local targetCount = math.floor(totalParts * (percent / 100))
    local currentCount = 0

    for part, data in pairs(originalMaterials) do
        if part and part.Parent then
            currentCount = currentCount + 1
            if currentCount <= targetCount then
                if enable then
                    part.Material = Enum.Material.SmoothPlastic
                    part.Reflectance = 0
                    part.CastShadow = false
                else
                    part.Material = data.Material
                    part.Reflectance = data.Reflectance
                    part.CastShadow = data.CastShadow
                end
            end
        end
    end
end

local NPC2_CENTER = Vector3.new(-2790,239,-1762)
local CurrentPoint = 1
local LastTeleport = 0
local LastClick = 0

local POINT_ANGLES = {
    math.rad(-90),
    math.rad(30),
    math.rad(150)
}

local function FindNPC2()
    local npc2 = workspace:FindFirstChild("NPC2",true)
    if not npc2 then return nil,nil end
    if npc2:IsA("Model") then
        local root = npc2:FindFirstChild("HumanoidRootPart") or npc2.PrimaryPart or npc2:FindFirstChildWhichIsA("BasePart")
        if root then return npc2,root end
    elseif npc2:IsA("BasePart") then
        return npc2,npc2
    end
    return nil,nil
end

local function GetCenterPoint(index)
    local distance = tonumber(_G.BossSettings.Mode3.Dist) or 17
    if distance < 0 then distance = 0 end
    local angle = POINT_ANGLES[index] or POINT_ANGLES[1]
    local offset = Vector3.new(math.cos(angle)*distance,0,math.sin(angle)*distance)
    return Vector3.new(NPC2_CENTER.X+offset.X,NPC2_CENTER.Y,NPC2_CENTER.Z+offset.Z)
end

local function TeleportToPoint(index)
    local Character = player.Character
    if not Character then return end
    local HRP = Character:FindFirstChild("HumanoidRootPart")
    if not HRP then return end

    local Boss,BossRoot = FindNPC2()
    if not BossRoot then return end

    local Target = GetCenterPoint(index)
    local LookAt = Vector3.new(BossRoot.Position.X,Target.Y,BossRoot.Position.Z)

    HRP.CFrame = CFrame.lookAt(Target,LookAt)
    HRP.AssemblyLinearVelocity = Vector3.zero
end

RunService.Heartbeat:Connect(function()
    if _G.SelectedBossMode ~= "Chế Độ 3" or not _G.AutoFarmBossCircle then return end

    local now = os.clock()
    local speed = tonumber(_G.BossSettings.Mode3.Speed) or 0.08
    if speed <= 0 then speed = 0.01 end
    if now - LastTeleport < speed then return end
    LastTeleport = now

    local Character = player.Character
    local HRP = Character and Character:FindFirstChild("HumanoidRootPart")
    if not HRP then return end

    local Boss,BossRoot = FindNPC2()
    if not BossRoot then return end

    TeleportToPoint(CurrentPoint)

    CurrentPoint += 1
    if CurrentPoint > 3 then
        CurrentPoint = 1
    end
end)

local ArenaRadius = 55

local CameraHeight = 60

local OrbitSpeed = 0.3

local currentAngle = 0

local cameraConnection = nil

local cameraPart = nil

local function getBoss2()

    local gh2 = workspace:FindFirstChild("GiangHo2")

    if not gh2 then return nil end

    local npcs = gh2:FindFirstChild("NPCs")

    if not npcs then return nil end

    return npcs:FindFirstChild("NPC2")

local function SetupArenaCamera(enabled)

    if enabled then

        if cameraConnection then

            cameraConnection:Disconnect()

            cameraConnection = nil

        end

        if cameraPart then

            cameraPart:Destroy()

            cameraPart = nil

        end

        

        cameraPart = Instance.new("Part")

        cameraPart.Name = "ArenaCameraAnchor"

        cameraPart.Size = Vector3.new(1, 1, 1)

        cameraPart.Transparency = 1

        cameraPart.CanCollide = false

        cameraPart.Anchored = true

        cameraPart.Parent = workspace

        

        workspace.CurrentCamera.CameraType = Enum.CameraType.Scriptable

        currentAngle = 0

        

        cameraConnection = RunService.RenderStepped:Connect(function(deltaTime)

            if enabled then

                currentAngle = currentAngle + (OrbitSpeed * deltaTime)

                

                local offsetX = math.cos(currentAngle) * ArenaRadius

                local offsetZ = math.sin(currentAngle) * ArenaRadius

                

                local currentCenter = ArenaCenter

                local liveBoss = getBoss2()

                if liveBoss and liveBoss:FindFirstChild("HumanoidRootPart") then

                    currentCenter = liveBoss.HumanoidRootPart.Position

                end

                

                local cameraPos = Vector3.new(

                    currentCenter.X + offsetX,

                    currentCenter.Y + CameraHeight,

                    currentCenter.Z + offsetZ

                )

                

                cameraPart.CFrame = CFrame.new(cameraPos, currentCenter)

                workspace.CurrentCamera.CFrame = cameraPart.CFrame

                workspace.CurrentCamera.CameraSubject = cameraPart

            end

        end)

    else

        if cameraConnection then

            cameraConnection:Disconnect()

            cameraConnection = nil

        end

        if cameraPart then

            cameraPart:Destroy()

            cameraPart = nil

        end

        workspace.CurrentCamera.CameraType = Enum.CameraType.Custom

        if Player.Character then

            workspace.CurrentCamera.CameraSubject = Player.Character:FindFirstChildOfClass("Humanoid")

        end

    end

end

local function getClosestPlayer()
    local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end

    local closest, shortest = nil, math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player then
            local char = p.Character
            local hrp  = char and char:FindFirstChild("HumanoidRootPart")
            local hum  = char and char:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                local dist = (myRoot.Position - hrp.Position).Magnitude
                if dist < shortest then
                    shortest = dist
                    closest  = p
                end
            end
        end
    end
    return closest
end

RunService.RenderStepped:Connect(function()
    pcall(function()
        local char = player.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        local hum  = char and char:FindFirstChildOfClass("Humanoid")

        -- 1. Speed (CFrame – không bị kick)
        if _G.PvpSpeedActive and hrp and hum and hum.MoveDirection.Magnitude > 0 then

            if not hrp:FindFirstChild("VND_SpinVelocity") then

                spin.Name                = "VND_SpinVelocity"
                spin.MaxTorque           = Vector3.new(0, math.huge, 0)
                spin.AngularVelocity     = Vector3.new(0, 40, 0)
                spin.Parent              = hrp
            end
        else
            if hrp then
                local sv = hrp:FindFirstChild("VND_SpinVelocity")
                if sv then sv:Destroy() end
            end
        end

        -- 3. Aim (không di chuyển camera)
        if _G.PvpAimNoCamActive and hrp then
            local target = getClosestPlayer()
            if target and target.Character then
                local tHrp = target.Character:FindFirstChild("HumanoidRootPart")
                if tHrp then
                    local lookAt = Vector3.new(tHrp.Position.X, hrp.Position.Y, tHrp.Position.Z)
                    hrp.CFrame   = CFrame.new(hrp.Position, lookAt)
                end
            end
        end
    end)
end)

RunService.Heartbeat:Connect(function()
    -- Tắt cả nếu không cần
    if not _G.PvpHitboxActive and not _G.PvpHealthActive then
        if next(activeHitboxes)   then clearAllHitboxes()    end
        if next(activeHealthGuis) then clearAllHealthGuis()  end
        return
    end

    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player then
            local pChar = p.Character
            local pHrp  = pChar and pChar:FindFirstChild("HumanoidRootPart")
            local pHead = pChar and pChar:FindFirstChild("Head")
            local pHum  = pChar and pChar:FindFirstChildOfClass("Humanoid")

            -- === HITBOX ===
            if _G.PvpHitboxActive and pHrp and pHum and pHum.Health > 0 then
                if pHrp.Size ~= Vector3.new(5, 5, 5) then
                    pcall(function()
                        pHrp.Size       = Vector3.new(5, 5, 5)
                        pHrp.CanCollide = false
                    end)
                end
                if not activeHitboxes[p.Name] then
                    local box           = Instance.new("BoxHandleAdornment")
                    box.Size            = Vector3.new(5, 6, 5)
                    box.Color3          = Color3.fromRGB(255, 50, 50)
                    box.Transparency    = 0.55
                    box.AlwaysOnTop     = true
                    box.Adornee         = pHrp
                    box.Parent          = ScreenGui
                    activeHitboxes[p.Name] = box
                end
            else
                if activeHitboxes[p.Name] then
                    pcall(function() activeHitboxes[p.Name]:Destroy() end)
                    activeHitboxes[p.Name] = nil
                end
                if pHrp then
                    pcall(function() pHrp.Size = Vector3.new(2, 2, 1) end)
                end
            end

            -- === ESP CARD ===
            if _G.PvpHealthActive and pHead and pHum and pHum.Health > 0 then
                if not activeHealthGuis[p.Name] then
                    createHealthGui(p, pHead)
                end
                updateHealthGui(activeHealthGuis[p.Name], pHum)
            else
                if activeHealthGuis[p.Name] then
                    pcall(function() activeHealthGuis[p.Name]:Destroy() end)
                    activeHealthGuis[p.Name] = nil
                end
            end
        end
    end
end)

local function EscapeRichText(Text)
    Text = tostring(Text or "")

    Text = string.gsub(Text,"&","&amp;")
    Text = string.gsub(Text,"<","&lt;")
    Text = string.gsub(Text,">","&gt;")
    Text = string.gsub(Text,'"',"&quot;")
    Text = string.gsub(Text,"'","&apos;")

    return Text
end

local function ColorToHex(Color)
    if not Color then
        return "#FFFFFF"
    end

    local R = math.floor(Color.R * 255 + 0.5)
    local G = math.floor(Color.G * 255 + 0.5)
    local B = math.floor(Color.B * 255 + 0.5)

    return string.format(
        "#%02X%02X%02X",
        R,
        G,
        B
    )
end

local function HexToColor(Hex)
    Hex = tostring(Hex or "")

    Hex = string.gsub(Hex,"#","")
    Hex = string.gsub(Hex,"%s+","")

    if #Hex ~= 6 then
        return nil
    end

    local R = tonumber(string.sub(Hex,1,2),16)
    local G = tonumber(string.sub(Hex,3,4),16)
    local B = tonumber(string.sub(Hex,5,6),16)

    if not R or not G or not B then
        return nil
    end

    return Color3.fromRGB(R,G,B)
end

local function NormalizeText(Text)
    Text = tostring(Text or "")
    Text = string.lower(Text)

    Text = string.gsub(Text,"<.->","")
    Text = string.gsub(Text,"%s+"," ")
    Text = string.gsub(Text,"^%s*(.-)%s*$","%1")

    return Text
end

-- =========================================================
-- FIND RANK UI
-- =========================================================

local function FindValidRankUI()
    local Character = LocalPlayer.Character

    if not Character then
        return nil
    end

    local ExactNames = {
        "RankUI",
        "NameTag",
        "Nametag",
        "NameTagUI",
        "RankGui",
        "RankGUI",
        "Overhead",
        "OverheadGui",
        "PlayerRank",
        "PlayerTag"
    }

    local Candidates = {}
    local Added = {}

    for _,Name in ipairs(ExactNames) do
        local Object

        pcall(function()
            Object = Character:FindFirstChild(Name,true)
        end)

        if Object then
            if Object:IsA("BillboardGui")
                or Object:IsA("SurfaceGui")
                or Object:IsA("Frame")
                or Object:IsA("Folder")
                or Object:IsA("Model") then

                if not Added[Object] then
                    Added[Object] = true
                    table.insert(Candidates,Object)
                end
            end
        end
    end

    local Descendants = {}

    pcall(function()
        Descendants = Character:GetDescendants()
    end)

    for _,Object in ipairs(Descendants) do
        if Object:IsA("BillboardGui")
            or Object:IsA("SurfaceGui") then

            if not Added[Object] then
                Added[Object] = true
                table.insert(Candidates,Object)
            end
        end
    end

    local Best
    local BestScore = -math.huge

    for _,Gui in ipairs(Candidates) do
        if IsLocalCharacterDescendant(Gui) then
            local Labels = GetLabels(Gui)

            if #Labels >= 3 then
                local Score = #Labels * 10
                local GuiName = NormalizeText(Gui.Name)

                if string.find(GuiName,"rank",1,true) then
                    Score = Score + 80
                end

                if string.find(GuiName,"name",1,true) then
                    Score = Score + 50
                end

                if string.find(GuiName,"tag",1,true) then
                    Score = Score + 40
                end

                if Score > BestScore then
                    BestScore = Score
                    Best = Gui
                end
            end
        end
    end

    return Best
end

-- =========================================================
-- RESOLVE LABELS
-- =========================================================

local function ResolveLabels(Gui)
    local Labels = GetLabels(Gui)

    if #Labels < 3 then
        return false
    end

    local UsernameLabel
    local UsernameScore = -math.huge

    for _,Label in ipairs(Labels) do
        local Score = ScoreUsernameLabel(Label)

        if Score > UsernameScore then
            UsernameScore = Score
            UsernameLabel = Label
        end
    end

    local RankLabel
    local RankScore = -math.huge

    for _,Label in ipairs(Labels) do
        if Label ~= UsernameLabel then
            local Score = ScoreRankLabel(Label)

            if Score > RankScore then
                RankScore = Score
                RankLabel = Label
            end
        end
    end

    local LevelLabel
    local LevelScore = -math.huge

    for _,Label in ipairs(Labels) do
        if Label ~= UsernameLabel
            and Label ~= RankLabel then

            local Score = ScoreLevelLabel(Label)

            if Score > LevelScore then
                LevelScore = Score
                LevelLabel = Label
            end
        end
    end

    if not UsernameLabel
        or not RankLabel
        or not LevelLabel then
        return false
    end

    labelUsername = UsernameLabel
    labelDisplay = RankLabel
    labelGroupRank = LevelLabel

    return true
end

-- =========================================================
-- UPDATE REFERENCES
-- =========================================================

local function UpdateReferences()
    local Character = LocalPlayer.Character

    if not Character then
        return false
    end

    if LastCharacter ~= Character then
        LastCharacter = Character
        RankUIReference = nil
        ClearReferences()
        StableCount = 0
    end

    if RankUIReference then
        if not IsLocalCharacterDescendant(RankUIReference) then
            RankUIReference = nil
            ClearReferences()
            StableCount = 0
        end
    end

    local Found = FindValidRankUI()

    if not Found then
        StableCount = 0
        UIReady = false
        return false
    end

    if RankUIReference == Found then
        StableCount = StableCount + 1
    else
        RankUIReference = Found
        StableCount = 1
        ClearReferences()
    end

    if StableCount < StableNeeded then
        return false
    end

    if not ResolveLabels(Found) then
        UIReady = false
        return false
    end

    UIReady = true
    return true
end

        or not labelDisplay
        or not labelGroupRank then
        return false
    end

    local Success = false

    -- =====================================================
    -- USERNAME
    -- =====================================================

    local UsernameText = GetInputText(BoxUser)

    if UsernameText == "" then
        if IsNonEmpty(customUsername) then
            UsernameText = tostring(customUsername)

        elseif IsNonEmpty(AppliedState.UsernameText) then
            UsernameText = tostring(AppliedState.UsernameText)

        else
            UsernameText = tostring(labelUsername.Text or "")
        end
    end

    -- =====================================================
    -- RANK
    -- =====================================================

    local RankText = GetInputText(BoxRank)

    if RankText == "" then
        if IsNonEmpty(customDisplay) then
            RankText = tostring(customDisplay)

        elseif IsNonEmpty(AppliedState.RankText) then
            RankText = tostring(AppliedState.RankText)

        else
            RankText = tostring(labelDisplay.Text or "")
        end
    end

    -- =====================================================
    -- LEVEL
    -- =====================================================

    local Level = GetLevel()

    -- =====================================================
    -- NAME
    -- =====================================================

    local Name = GetName()

    -- =====================================================
    -- GROUP TEXT
    -- =====================================================

    local GroupText = ""
    local GroupRichText = true

    if Level ~= "" or Name ~= "" then

        GroupText = BuildLevelNameText(
            Level,
            Name,
            LevelColor,
            NameColor
        )

        GroupRichText = true

    elseif IsNonEmpty(AppliedState.GroupText) then

        -- Không có input mới:
        -- giữ nguyên GroupText của lần Apply trước.

        GroupText = tostring(
            AppliedState.GroupText
        )

        GroupRichText =
            AppliedState.GroupRichText == true

    else

        -- Lần Apply đầu tiên nhưng Level/Name để trống:
        -- giữ nguyên Text gốc của game.

        pcall(function()
            GroupText = tostring(
                labelGroupRank.Text or ""
            )

            GroupRichText =
                labelGroupRank.RichText == true
        end)
    end

    -- =====================================================
    -- APPLY USERNAME
    -- =====================================================

    if IsAlive(labelUsername)
        and IsLocalCharacterDescendant(labelUsername) then

        pcall(function()
            labelUsername.RichText = false
            labelUsername.Text = UsernameText
            labelUsername.TextColor3 = UsernameColor
            Success = true
        end)
    end

    -- =====================================================
    -- APPLY RANK
    -- =====================================================

    if IsAlive(labelDisplay)
        and IsLocalCharacterDescendant(labelDisplay) then

        pcall(function()
            labelDisplay.RichText = false
            labelDisplay.Text = RankText
            labelDisplay.TextColor3 = RankColor
            Success = true
        end)
    end

    -- =====================================================
    -- APPLY LEVEL + NAME
    -- =====================================================

    if IsAlive(labelGroupRank)
        and IsLocalCharacterDescendant(labelGroupRank) then

        pcall(function()
            labelGroupRank.RichText = GroupRichText
            labelGroupRank.Text = GroupText
            Success = true
        end)
    end

    -- =====================================================
    -- SAVE EXACT STATE
    -- =====================================================

    if Success then
        SaveAppliedState(
            UsernameText,
            RankText,
            Level,
            Name,
            GroupText,
            GroupRichText
        )
    end

    return Success
end

-- =========================================================
-- RECOLOR RICHTEXT
-- =========================================================
-- Không tạo lại Lv/Name.
-- Chỉ thay màu các <font color="..."> hiện có.
-- Nếu Text chưa có font tag thì bọc toàn bộ text.

local function RecolorRichText(Text,Color)
    Text = tostring(Text or "")

    local Hex = ColorToHex(Color)

    local Changed = false

    local Result = string.gsub(
        Text,
        '(<font%s+color%s*=%s*")#[%x]+(")',
        '%1' .. Hex .. '%2'
    )

    if Result ~= Text then
        Changed = true
        Text = Result
    end

    if not Changed then
        local Result2 = string.gsub(
            Text,
            '(<font%s+color%s*=%s*")([^"]+)(")',
            '%1' .. Hex .. '%3'
        )

        if Result2 ~= Text then
            Changed = true
            Text = Result2
        end
    end

    if not Changed and Text ~= "" then
        Text =
            '<font color="' ..
            Hex ..
            '">' ..
            Text ..
            '</font>'
    end

    return Text
end

-- =========================================================
-- PICKER
-- =========================================================

local function UpdateSV(X,Y)
    X = math.clamp(X,0,1)
    Y = math.clamp(Y,0,1)

    CurrentSaturation = X
    CurrentValue = 1 - Y

    if SVMarker then
        SVMarker.Position =
            UDim2.new(
                X,0,
                Y,0
            )
    end

    SetTargetColor(
        Color3.fromHSV(
            CurrentHue,
            CurrentSaturation,
            CurrentValue
        )
    )
end

local function UpdateHue(Y)
    Y = math.clamp(Y,0,1)

    CurrentHue = 1 - Y

    if HueMarker then
        HueMarker.Position =
            UDim2.new(
                0,-3,
                Y,-3
            )
    end

    SetTargetColor(
        Color3.fromHSV(
            CurrentHue,
            CurrentSaturation,
            CurrentValue
        )
    )

    if SVFrame then
        SVFrame.BackgroundColor3 =
            Color3.fromHSV(
                CurrentHue,
                1,
                1
            )
    end
end

-- =========================================================
-- PICKER INPUT
-- =========================================================

local function ConnectPickerObject(
    Object,
    Callback
)
    Object.InputBegan:Connect(
        function(Input)

            local Valid =
                Input.UserInputType ==
                    Enum.UserInputType.MouseButton1
                or
                Input.UserInputType ==
                    Enum.UserInputType.Touch

            if not Valid then
                return
            end

            local function Update(InputObject)

                if not ColorPicker
                    or not ColorPicker.Visible then
                    return
                end

                local Position =
                    Object.AbsolutePosition

                local Size =
                    Object.AbsoluteSize

                if Size.X <= 0
                    or Size.Y <= 0 then
                    return
                end

                Callback(
                    InputObject,
                    Position,
                    Size
                )
            end

            Update(Input)

            StopPicker()

            PickerConnection =
                UserInputService.InputChanged:Connect(
                    function(Move)

                        local ValidMove =
                            Move.UserInputType ==
                                Enum.UserInputType.MouseMovement
                            or
                            Move.UserInputType ==
                                Enum.UserInputType.Touch

                        if ValidMove then
                            Update(Move)
                        end
                    end
                )
        end
    )
end

-- =========================================================
-- DRAG
-- =========================================================

local function MakeDraggable(
    Frame,
    DragObject
)
    DragObject.InputBegan:Connect(
        function(Input)

            local Valid =
                Input.UserInputType ==
                    Enum.UserInputType.MouseButton1
                or
                Input.UserInputType ==
                    Enum.UserInputType.Touch

            if not Valid then
                return
            end

            local Dragging = true

            local DragStart =
                Input.Position

            local StartPosition =
                Frame.Position

            if MainDragConnection then
                pcall(function()
                    MainDragConnection:Disconnect()
                end)

                MainDragConnection = nil
            end

            MainDragConnection =
                UserInputService.InputChanged:Connect(
                    function(Move)

                        if not Dragging then
                            return
                        end

                        local ValidMove =
                            Move.UserInputType ==
                                Enum.UserInputType.MouseMovement
                            or
                            Move.UserInputType ==
                                Enum.UserInputType.Touch

                        if not ValidMove then
                            return
                        end

                        local Delta =
                            Move.Position -
                            DragStart

                        Frame.Position =
                            UDim2.new(
                                StartPosition.X.Scale,
                                StartPosition.X.Offset +
                                    Delta.X,

                                StartPosition.Y.Scale,
                                StartPosition.Y.Offset +
                                    Delta.Y
                            )
                    end
                )

            Input.Changed:Connect(
                function()

                    if Input.UserInputState ==
                        Enum.UserInputState.End then

                        Dragging = false

                        if MainDragConnection then
                            pcall(function()
                                MainDragConnection:Disconnect()
                            end)

                            MainDragConnection = nil
                        end
                    end
                end
            )
        end
    )
end

    function()

        while true do

            task.wait(0.75)

            local Character =
                LocalPlayer.Character

            if Character then

                if LastCharacter ~= Character then

                    LastCharacter = Character
                    RankUIReference = nil

                    ClearReferences()

                    StableCount = 0
                end

                if RankUIReference then

                    if not IsLocalCharacterDescendant(
                        RankUIReference
                    ) then

                        RankUIReference = nil

                        ClearReferences()

                        StableCount = 0
                    end
                end

                if not UIReady then
                    UpdateReferences()
                end

            else

                UIReady = false

            end
        end
    end
)

Tabs.Main4:AddButton({
    Title = "Mua tất cả các lựa chọn",
    Callback = function()
        local hasSelected = false

        -- Mua Món ăn nếu có chọn
        if _G.SelectedFood then
            buyItemEvent:FireServer(_G.SelectedFood)
            Fluent:Notify({Title = "Mua Món ăn", Content = "Đang mua: " .. _G.SelectedFood, Duration = 3})
            hasSelected = true
        end
            
        -- Mua Vật phẩm nếu có chọn
        if _G.SelectedItem then
            buyItemEvent:FireServer(_G.SelectedItem)
            Fluent:Notify({Title = "Mua Vật phẩm", Content = "Đang mua: " .. _G.SelectedItem, Duration = 3})
            hasSelected = true
        end
            
        -- Mua Xe nếu có chọn
        if _G.SelectedVehicle then
            remoteEvent:FireServer(_G.SelectedVehicle)
            Fluent:Notify({Title = "Mua Xe", Content = "Đang mua: " .. _G.SelectedVehicle, Duration = 3})
            hasSelected = true
        end
            
        -- Thông báo nếu không có gì được chọn
        if not hasSelected then
            Fluent:Notify({Title = "Lỗi", Content = "Bạn chưa chọn bất kỳ món nào để mua!", Duration = 3})
        end
    end
})

-- 🚀 ĐỒNG BỘ HÓA TỰ ĐỘNG KHI VÀO GAME
task.defer(function()
    while not Fluent.Options or not Fluent.Options["UseConfigSystemToggle"] or not Fluent.Options["SelectConfigDropdown"] do
        task.wait(0.2)
    end
    
    if isfile(GlobalSettingsFile) then
        pcall(function()
            local data = HttpService:JSONDecode(readfile(GlobalSettingsFile))
            if data and data["SelectedConfigName"] then
                _G.SelectedConfigName = data["SelectedConfigName"]
                _G.UseConfigSystem = data["UseConfigSystem"] or false
                
                Fluent.Options["UseConfigSystemToggle"]:SetValue(_G.UseConfigSystem)
                Fluent.Options["SelectConfigDropdown"]:SetValue(_G.SelectedConfigName)
                
                if _G.UseConfigSystem then
                    loadSelectedConfig(_G.SelectedConfigName)
                    Fluent:Notify({Title = "VND💵 System", Content = "Đã tự động nạp & kích hoạt: " .. _G.SelectedConfigName, Duration = 5})
                end
            end
        end)
    end
end)
